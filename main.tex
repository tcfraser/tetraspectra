\section{Non-linear inequalities}

\begin{lem}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices, then
    \begin{equation}
        \sum_{j}\lambda_j(A+B)^2 + \sum_{j}\lambda_j(B+C)^2 \leq \sum_{j}\lambda_j(B)^2 + \sum_{j}\lambda_j(A+B+C)^2.
    \end{equation}
\end{lem}
\begin{proof}
    Note that $\sum_{j}\lambda_{j}(X)^2  = \Tr[X^2]$ for any Hermitian operator $X$. Therefore,
    \begin{align}
        \begin{split}
            & \Tr[(B)^2] + \Tr[(A+B+C)^2] - \Tr[(A+B)^2] - \Tr[(B+C)^2] \\
            &\quad= 
            (\Tr[B^2])
            +(\Tr[A^2] + \Tr[B^2] + \Tr[C^2] + 2\Tr[AB] + 2\Tr[BC] + 2 \Tr[AC])\\
            &\qquad\qquad-(\Tr[A^2] + \Tr[B^2] + 2\Tr[AB])
            -(\Tr[B^2] + \Tr[C^2] + 2\Tr[BC])\\
            &\quad= 2 \Tr[AC].
        \end{split}
    \end{align}
    Since $A,C\geq 0$ are both positive semidefinite, we also have $\Tr[AC] \geq 0$ which proves the claim.
\end{proof}

\begin{thm}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices, then
    \begin{align}
        &\sum_{j}\lambda_j(A+B)\log \lambda_j(A+B)+ \sum_{j}\lambda_j(B+C)\log \lambda_j(B+C) \\
        &\qquad\leq \sum_{j}\lambda_j(B)\log\lambda_j(B) + \sum_{j}\lambda_j(A+B+C)\log\lambda_j(A+B+C).
    \end{align}
\end{thm}

\subsection{Horn's problem \& generalizations}
\begin{itemize}
     \item \Citeauthor{horn1962eigenvalues}'s original paper where they conjectured the form of a necessary and sufficient set of inequalities via an inductive scheme for increasing dimension \cite{horn1962eigenvalues}
    \item An excellent expository overview of the big picture by \citeauthor{knutson2000symplectic} \cite{knutson2000symplectic}
    \item An excellent introduction to these ideas is the recent masters thesis on various aspects of additive and multiplicative Horn problems in connection to Vinnikov curves and the Lax conjecture due to \citeauthor{hess2020examination} and supervised by S. Kuhlmann and M. Infusino \cite{hess2020examination}
    \item Quantum-info style proof of Littlewood-Richardson coefficient connection due to \citeauthor{christandl2008quantum} \cite{christandl2008quantum}
    \item \Citeauthor{coquereaux2018orbital} explictly outline the link between the orbital measures (i.e. the induced distributions in the sense of Klyachko's random walk approach) and the magnitude of the stretched Littlewood-Richardson coefficients \cite{coquereaux2018orbital}. See also the papers \cite{coquereaux2020horn,coquereaux2019revisiting,zuber2018horn}.
    \item A recent article by \citeauthor{mcswiggen2021harish} is an excellent introduction to the Harish-Chandra integral \cite{mcswiggen2021harish}. The proof of the Harish-Chandra formula provide by \citeauthor{mcswiggen2021harish} using the heat equation methods from \citeauthor{fegan1978heat}'s 1978 paper \cite{fegan1978heat} seems very promising for deriving a connection to the character formula for the norm of the recoupling operator!
    \item \citeauthor{speyer2005horn} \cite{speyer2005horn}
    \item Generalizations of Horn's problem to the product of unitary matrices \cite{agnihotri1997eigenvalues,belkale2008quantum,belkale2016multiplicative}
    \item \citeauthor{bhatia2001linear} \cite{bhatia2001linear}
    \item History account offered by \citeauthor{daftuar2005quantum} \cite{daftuar2005quantum}, and technical review article by \citeauthor{fulton2000eigenvalues} \cite{fulton2000eigenvalues}.
    \item \citeauthor{berline2018horn} a pedagogical account of the triple role played by Horn's inequalities in characterizing the existence of invariants vectors in tensor products, the eigenvalues of sums of Hermitian matrices, and intersections of Schubert varieties \cite{berline2018horn}
    \item \citeauthor{fulton1997young} provides a comphrensive treatment of the connection between representation theory of $\GL(d)$ and the geometry and (co)homology of the complex Grassmannians \cite{fulton1997young}
    \item Prakash Belkale has contributed greatly to the theory surrounding Horn-type problems: for example, in \cite{belkale2004invariant}, Belkale provides a \textit{geometric} proof of the equivalence between (i) the structure coefficients of the representation theory of $\GL(d)$ (i.e. the LR-coefficients $c_{\mu\nu}^{\lambda}$), and (ii) the structure coefficients of the cohomology ring of the Grassmannian $\mathrm{Gr}(d, n)$ of $d$-dimensional subspaces in $\mathbb C^{d}$. This proof is now complementary to the previously known algebraic proofs which rely on knowledge of the algebra of symmetric functions. This geometric approach taken by Belkale, which emphasizes the tangent spaces of the involved Schubert varieties, is later used by Belkale to provide a geometric proof of the saturation conjecture \cite{belkale2002geometric}.
\end{itemize}

\subsection{\texorpdfstring{$6j$}{6j} symbols}

    \item A great summary by \citeauthor{roberts2002asymptotics} \cite{roberts2002asymptotics}; also the original result for the asymptotics of the $6j$ symbols due to  \citeauthor{roberts1999classical} \cite{roberts1999classical}
    \item The masters essay/thesis of \citeauthor{backens2010sixjsymbols} (\textcolor{red}{TODO: check if the pdf Matthias sent to me is the final version}) \cite{backens2010sixjsymbols}
    \item The spectral quantum marginal problem and the $6j$ symbol for the symmetric group \citeauthor{christandl2018recoupling} \cite{christandl2018recoupling}
    \item Usage of Wigner's $6j$ symbol to compute asymptotic errors rates for the promise problem of distinguishing between two possible ``change points'' for a sequence of unknown pure states to transition from one unknown type to another \cite{akimoto2011discrimination}
    \item See also \Citeauthor{coquereaux2023integer}'s later paper which promotes the alternative, albeit equivalent, usage of the integer-valued $\mathrm{gon}$ and $\mathrm{tet}$ functions in place of the rational-valued theta (triangular) and irrational-valued 6j symbols \cite{coquereaux2023integer}. Most relevant for this project is \cite[Section IV D]{coquereaux2023integer} which discusses the apparent lack of substantial generalization from $\SU(2)$ to $\SU(N)$ (\textit{``apart from very few special cases (low dimension of irreps and no multiplicity) worked out by physicists at the time of the eightfold way, we are not aware of any published general result''}).
    \item The recent paper by \citeauthor{alekseev2020multiplicity} (different Alekseev!) covers the case of some multiplicity free formulas for the quantum $6j$ symbols for $\U_q(\mathrm{sl}_{N})$ \cite{alekseev2020multiplicity}
    \item \citeauthor{keppeler2024wigner} recently calculated $\SU(d)$ $6j$ symbols with adjoint representations \cite{keppeler2024wigner} and a bit earlier \cite{alcock2023wigner} calculated $\SU(d)$ $6j$ symbols with at least two fundamental representations \cite{alcock2023wigner}.
    \item Representation theoretic proof of the Regge symmetries due to \citeauthor{boalch2007regge} \cite{boalch2007regge}
    \item Asymptotics of quantum $6j$ symbols \citeauthor{taylor2006sixj} \cite{taylor2006sixj}
    \item The integral formula for the square of the $\SU(2)$ $6j$ symbol can be performed by an appropriate change of variables \cite{freidel2003asymptotics}. See also other papers by Laurent Freidel and collaborators that are relavent \cite{freidel2000simple,freidel20076j,freidel2010holomorphic}.
    \item \citeauthor{alcockspecial} consider the case of $6j$ symbols for $\SU(N)$ for wherein two opposing edges are labelled by fundamental irreps of $\SU(N)$ so that all vertices are multiplicity one (or zero) \cite{alcockspecial} \textcolor{blue}{TC: They also freely use the birdtrack notation which I enjoy.}
    \item Only the tetrahedral symmetry of the $\U_q(\SL(N))$ $6j$-symbol is known, the counterpart of the Regge symmetry is still unknown, although \citeauthor{morozov2018new} conjectures the generalized Regge symmetry may be related to the eigenvalue conjecture, just as it is in the case when $N=2$ \cite{morozov2018new}.
    \item The graphical notion, specifically the use of an additional edge for multiplicity spaces can has been used for a long time, as early as \cite{agrawala1968graphical,moussouris1984quantum}.
    \item The original Ponzano-Regge paper \cite{ponzano1968semiclassical}
    \item \citeauthor{gurau2008ponzano} derives the Ponzano-Regge asymptotic formula directly using Stirling's approximation applied to Racah's single-sum formula \cite{gurau2008ponzano}
    \item \citeauthor{bartlett2023coherent} work with so-called \textit{coherent loop states} or \textit{Bohr-Sommerfeld states} to develop some semiclassical results concerning the inner products between such states (see also the talk on Youtube) \cite{bartlett2023coherent}
    \item \citeauthor{aquilanti2012semiclassical} provides an excellent introduction to subject of classical $6j$ symbols, their history, diagrammatics, presentations and asymptotics \cite{aquilanti2012semiclassical}
    \item \citeauthor{marzuoli2005computing} connects quantum computing models to the recoupling theory of $\SU(2)$ angular momentum and also considers the semiclassical limit \cite{marzuoli2005computing}
    \item \citeauthor{artamonov2023classical} derives an explicit formula for the $d=3$ case \cite{artamonov2023classical}, and later to arbitrary $d$ \cite{artamonov2024calculation}.
    \item There is a series of papers involving Alexey Sleptsov, starting with a computation of quantum $6j$ symbols for generic dimension $d$ but with symmetric representations in terms of certain hypergeometric polynomials \cite{mironov2017sixj}, then with a discussion of the eigenvalue hypothesis and its potential connection to generalized Regge symmetries \cite{morozov2018new}, then with a calculation of all multiplicity free quantum $6j$ symbols for arbitrary dimension \cite{alekseev2020multiplicity}, and then finally with the discovery of novel ``tug-the-hook'' symmetries for quantum $6j$ symbols for all $d$ \cite{lanina2023tug}.
\end{itemize}

The substitution mentioned in \cite{aiston1998idempotents} is the following
        \begin{equation*}
            s = e^{h/2}, \qquad
            v = e^{-dh/2} = s^{-d}, \qquad

            q = e^{h} = s^2.

    \begin{equation}
        [k] \coloneqq \frac{s^{+k} - s^{-k}}{s^{+1} - s^{-1}} = \frac{q^{+k/2} - q^{-k/2}}{q^{+1/2} - q^{-1/2}}.
    \end{equation}

\subsection{Eigenvalue inequalities}
\begin{itemize}
    \item The general subject of multivariate trace inequalities involving exponentials of sums of Hermitian and products of their exponentials is, of course, intimately related. See for instance the paper by \citeauthor{sutter2017multivariate} \cite{sutter2017multivariate}.
    \item Also the article \cite{forrester2014golden} by \citeauthor{forrester2014golden} which describes the connection between Thompson's inequality and hyperbolic geometry quite explicitly.
     \item \citeauthor{friedland2013submodular} considers submodular functions associated to operator monotones evaluated on different principal submatrices of a single Hermitian matrix \cite{friedland2013submodular}. While it is not clear to be yet whether or not one can relate partial sums of matrices to principal submatrices, \cite{friedland2013submodular} references a paper by \citeauthor{bouhtou2008optimization} \cite{bouhtou2008optimization} which proves the following family of inequalities for all positive semidefinite $A,B,C \geq 0$ (and $p \in [0,1]$) \cite[Lemma 5.3]{bouhtou2008optimization}:
        \begin{equation}
            \sum_{j}\lambda_j(A+B)^p + \sum_{j}\lambda_j(B+C)^p \geq \sum_{j}\lambda_j(B)^p + \sum_{j}\lambda_j(A+B+C)^p.
        \end{equation}
    \item \citeauthor{audenaert2010strongly} consider a similar form of ``strongly subadditive'' functions \cite{audenaert2010strongly}
    \item For multivariate inequalities based on idempotents of the symmetric group algebra see \citeauthor{huber2020positive}'s paper \cite{huber2020positive}.
    \item For spectral inequalities derived using semidefinite programming, see \citeauthor{huber2022refuting}'s paper \cite{huber2022refuting}.
\end{itemize}

\subsection{Uncategorized}
\begin{itemize}
    \item Elementary introduction to toy models of three-dimensional quantum space \cite{barrett2003geometrical}
    \item \citeauthor{regge2000discrete} \cite{regge2000discrete} a relatively recent review of discrete approaches to gravity. 
        The essential idea, in constrast to the differential geometry of general relativity, is to model curvature discretely instead of continuously. 
        In particular, one can think of a triangulation of the surface of a sphere; the curvature of the sphere is encoded in the triangulation when a configuration of three (or more) triangles meet at a common vertex with sum of interior angles deviating from $\pi$.
        It is in this sense that \citeauthor{regge2000discrete} refers to these types of models as ``piecewise-flat''.
        Contained is an overview of various attempts to generalize discrete theories of three-dimensional gravity to four dimensions \cite[Section III]{regge2000discrete}.
\end{itemize}
