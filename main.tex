\section{Tetrahedra \& Hermitian matrices}

\subsection{Definition}
\begin{prob}[Tetrahedral eigenvalues]
    \label{prob:tetrahedral_problem}
    A sextuple $(\alpha,\beta,\gamma,\delta,\epsilon,\phi) \in (\eigs{d})^{\times 6}$ of candidate eigenvalues are said to be \defnsty{tetrahedral} if there exists a corresponding sextuple $(A,B,C,D,E,F)$ of $d\times d$ Hermitian matrices satisfying (i) the four linear constraints
    \begin{equation}
        \label{eq:tetrahedral_linear_constraints}
        A + B = C, \quad
        B + D = F, \quad
        D + C = E, \quad
        A + F = E,
    \end{equation}
    and (ii) the six eigenvalue constraints
    \begin{equation}
        \label{eq:tetrahedral_eigenvalue_constraints}
        \lambda(A) = \alpha, \quad
        \lambda(B) = \beta, \quad
        \lambda(C) = \gamma, \quad
        \lambda(D) = \delta, \quad
        \lambda(E) = \epsilon, \quad
        \lambda(F) = \phi.
    \end{equation}
\end{prob}

\subsection{Symmetries}
\begin{lem}[Shifting]
     Let $(\tilde \alpha, \tilde \beta, \tilde \gamma, \tilde \delta, \tilde \epsilon, \tilde \phi)$ be a given sextuple of candidate eigenvalues and let $x,y,z \in \mathbb R$ be given scalars and define \textit{shifted} candidate eigenvalues by the formula
    \begin{equation}
        \label{eq:shifted_tetrahedral_eigenvalues}
        \begin{alignedat}{3}
            &\alpha \coloneqq \tilde \alpha + x, \qquad
            &&\beta \coloneqq \tilde \beta + y, \qquad
            &&\gamma \coloneqq \tilde \gamma + x+y, \\
            &\delta \coloneqq \tilde \delta + z, \qquad
            &&\epsilon \coloneqq \tilde \epsilon + x + y + z, \qquad
            &&\phi \coloneqq \tilde \phi + y + z.
        \end{alignedat}
    \end{equation}
    Then the shifted eigenvalues $(\alpha,\beta,\gamma,\delta,\epsilon,\phi)$ are tetrahedral if and only if the unshifted eigenvalues are tetrahedral.
    Moreover if the unshifted eigenvalues are tetrahedral and $(x,y,z)$ are sufficiently large, i.e.
    \begin{equation}
        x \geq \min(\tilde \alpha), \qquad
        y \geq \min(\tilde \beta), \qquad
        z \geq \min(\tilde \delta),
    \end{equation}
    then the shifted spectra are both tetrahedral and non-negative.
\end{lem}

\subsection{The \texorpdfstring{$\boldsymbol{d=2}$}{d=2} case}
In the context of the $d=2$ tetrahedral eigenvalue problem, each traceless $2 \times 2$ matrix, $X$, may be identified with a three-dimensional vector $(x_1, x_2, x_3) \in \mathbb R^{3}$ using the standard Pauli-matrix basis:
\begin{equation}
    X = x_1 \sigma_1 + x_2 \sigma_2 + x_3 \sigma_3 = \begin{pmatrix} x_3 & x_1-ix_2 \\ x_1+ix_2 & -x_3 \end{pmatrix}.
\end{equation}
The eigenvalues of $X$ are then determined by the Euclidean length of the vector $(x_1, x_2, x_3)$:
\begin{equation}
    \lambda(X) = \lambda\begin{pmatrix} x_3 & x_1-ix_2 \\ x_1+ix_2 & -x_3 \end{pmatrix} = (\norm{x}, -\norm{x}), \qquad \text{where}, \qquad \norm{x} = \sqrt{x_1^2 + x_2^2 + x_3^2}.
\end{equation}

Now, for the special case of the tetrahedral eigenvalue constraint problem, one can assume the sextuple of eigenvalues are determined by a sextuple of non-negative real numbers $(a,b,c,d,e,f) \in \mathbb R_{\geq 0}^{6}$ such that
\begin{equation}
    \label{eq:traceless_two_dim_tetra}
    \alpha = (a, -a), \quad
    \beta = (b, -b), \quad
    \gamma = (c, -c), \quad
    \delta = (d, -d), \quad
    \epsilon = (e, -e), \quad
    \phi = (f, -f).
\end{equation}
Using this parameterization, it can be shown that the constraints imposed in \cref{prob:tetrahedral_problem} precisely describe the space of all possible edge-lengths $(a,b,c,d,e,f)$ of a genuine Euclidean tetrahedron in $\mathbb R^{3}$ \cite{richardson1902trigonometry}; namely, the conjunction of a quadruplet of triangle-inequalities,
\begin{equation}
    \label{eq:quadruplet_triangle_constraints}
    \abs{a-b} \leq c \leq a+b, \qquad
    \abs{b-d} \leq f \leq b+d, \qquad
    \abs{d-c} \leq e \leq d+c, \qquad
    \abs{a-f} \leq e \leq a+f,
\end{equation}
together with a single non-linear constraint of the form
\begin{equation}
    \label{eq:volume_positive_constraint}
    D(a,b,c,d,e,f) = \frac{1}{288}\det 
    \begin{pmatrix}
        0 & 1 & 1 & 1 & 1 \\ 
        1 & 0 & a^2 & e^2 & c^2 \\ 
        1 & a^2 & 0 & f^2 & b^2 \\ 
        1 & e^2 & f^2 & 0 & d^2 \\ 
        1 & c^2 & b^2 & d^2 & 0
    \end{pmatrix}
    \geq 0.
\end{equation}
The quantity $D(a,b,c,d,e,f)$ is known as a Cayley-Menger determinant and, when positive, is equal to square of the volume of the corresponding tetrahedron \cite{sitharam2018handbook}.
\Cref{fig:tetrahedron_moduli_space_example} depicts a three-dimensional projection of the solution space described by \cref{eq:quadruplet_triangle_constraints} and \cref{eq:volume_positive_constraint}.

\subsection{Comparison with the C\c{S}W inequality}
At this stage, it is worth comparing the inequality in \cref{eq:SSA_like_ineq} to a similar inequality derived by \citeauthor{christandl2018recoupling} \cite{christandl2018recoupling}, which holds for tetrahedral eigenvalues $(\alpha,\beta,\gamma,\delta,\epsilon,\phi)$ which are non-negative and sufficiently small in the sense that $\lambda_1 + \abs{\lambda} \leq 1$:
\begin{equation}
    \label{eq:SSA_like_CSW}
    h(\abs{\gamma}) 
    + \abs{\gamma} S(\bar \gamma)
    +h(\abs{\phi}) 
    + \abs{\phi} S(\bar \phi)  
    -h(\abs{\beta}) 
    - \abs{\beta} S(\bar \beta)  
    -h(\abs{\epsilon}) 
    - \abs{\epsilon} S(\bar \epsilon)  
    \geq 0.
\end{equation}
The inequality appearing in \cref{eq:SSA_like_ineq} is equivalent to the following by \cref{eq:little_s_relation}:
\begin{equation}
    \label{eq:SSA_like_ineq_equiv}
    -\abs{\gamma}\log\abs{\gamma}
    + \abs{\gamma} S(\bar \gamma)
    -\abs{\phi}\log\abs{\phi}
    + \abs{\phi} S(\bar \phi)  
    +\abs{\beta}\log\abs{\beta}
    - \abs{\beta} S(\bar \beta)  
    +\abs{\epsilon}\log\abs{\epsilon}
    - \abs{\epsilon} S(\bar \epsilon)  
    \geq 0.
\end{equation}
The difference between \cref{eq:SSA_like_CSW} and \cref{eq:SSA_like_ineq_equiv} is captured by a formal substitution replacing the binary entropy $h(\abs{\lambda})$ with the expression $-\abs{\lambda}\log \abs{\lambda}$.
We note that it is possible to derive \cref{eq:SSA_like_ineq_equiv} from \cref{eq:SSA_like_ineq_equiv} by a downscaling argument.
Specifically, while the set of tetrahedral eigenvalues is closed under scaling all eigenvalues by a non-negative $r$,
\begin{equation}
    \label{eq:tetrahedral_scaling}
    (\alpha,\beta,\gamma,\delta,\epsilon,\phi) \mapsto (r\alpha,r\beta,r\gamma,r\delta,r\epsilon,r\phi),
\end{equation}
the form and positivity of \cref{eq:SSA_like_CSW} is \textit{not} invariant under scaling by $r$.
Define $q_{r}$ by the formula
\begin{equation}
    q_{r} \coloneqq r^{-1}\left(
        h(\abs{r\gamma}) 
        +h(\abs{r\phi}) 
        -h(\abs{r\beta}) 
        -h(\abs{r\epsilon}) 
    \right).
\end{equation}
Letting $r \in (0,1)$ be positive and small, we see that
\begin{equation}
    r^{-1}h(\abs{r\lambda}) = \abs{\lambda}(1-\log r) - \abs{\lambda} \log \abs{\lambda} + O(r^2)
\end{equation}
which implies the following limit holds when $\abs{\gamma} + \abs{\phi} = \abs{\beta} + \abs{\epsilon}$:
\begin{equation}
    q_0 \coloneqq 
    \lim_{r \to 0} q_{r}  
    =
    -\abs{\gamma}\log\abs{\gamma}
    -\abs{\phi}\log\abs{\phi}
    +\abs{\beta}\log\abs{\beta}
    +\abs{\epsilon}\log\abs{\epsilon}.
\end{equation}
In summary, by scaling down the eigenvalues in \cref{eq:SSA_like_CSW} by $r \in (0,1)$ according to \cref{eq:tetrahedral_scaling} and then scaling up \cref{eq:SSA_like_CSW} by multiplication by $r^{-1}$ and taking the limit as $r \to \infty$, one recovers \cref{eq:SSA_like_ineq_equiv}.
Moreover, it can be shown that $q_{r} \geq q_{0}$ for all $r$ and thus \cref{eq:SSA_like_ineq_equiv} is strictly stronger than \cref{eq:SSA_like_CSW}.
To prove this, we see that the trace equalities imply
\begin{align}
    \begin{split}
        q_{r}-q_{0}
        &= r^{-1}(1-r\abs{\beta})\log(1-r\abs{\beta})+r^{-1}(1-r\abs{\epsilon})\log(1-r\abs{\epsilon})\\
        &\qquad\qquad-r^{-1}(1-r\abs{\gamma})\log(1-r\abs{\gamma})-r^{-1}(1-r\abs{\phi})\log(1-r\abs{\phi}).
    \end{split}
\end{align}
Approximating $-(1-x)\log(1-x)$ by a Taylor series for $x \approx 0$,
\begin{equation}
    -(1-x)\log(1-x) = x - \frac{x^2}{2} + O(x^3),
\end{equation}
produes
\begin{equation}
    q_{r} - q_{0} = \frac{r}{2}(\abs{\beta}^2 + \abs{\epsilon}^{2} - \abs{\gamma}^{2} - \abs{\phi}^{2}) + O(r^2) = r \abs{\alpha} \abs{\delta} + O(r^2)
\end{equation}
which implies $q_{r} > q_0$ and least for sufficiently small $r$.
To prove $q_{r} > q_0$ holds for all $r$, by continuity, it therefore suffices to show that the gradient of $q_{r} - q_{0}$ with respect to $r$ never changes sign for $r > 0$.
\begin{align}
    \begin{split}
        \partial_{r} (q_{r}-q_{0})
        &= \frac{1}{r^2} \left(\log(1-r\abs{\gamma})+\log(1-r\abs{\phi})-\log(1-r\abs{\beta})-\log(1-r\abs{\epsilon}) \right)\\
        &= \frac{1}{r^2} \log\left(\frac{(1-r\abs{\gamma})(1-r\abs{\phi})}{(1-r\abs{\beta})(1-r\abs{\epsilon})} \right).
    \end{split}
\end{align}
Thus the gradient $\partial_{r} (q_{r}-q_{0})$ vanishes only if the argument of the above logarithm is one, or equivalently when
\begin{equation}
    0 = (1-r\abs{\gamma})(1-r\abs{\phi})-(1-r\abs{\beta})(1-r\abs{\epsilon}) = r^2(\abs{\gamma}\abs{\phi} - \abs{\epsilon}\abs{\beta}) = r^2 \abs{\alpha}\abs{\delta}.
\end{equation}
As we are implicitly assuming $\abs{\alpha} \neq 0$ and $\abs{\delta} \neq 0$, we see that the gradient of $q_{r} - q_{0}$ never changes when $r>0$ and thus $q_{r}-q_{0}$ is increasing with increasing $r > 0$.

\section{Submodular and supermodular inequalities on positive semidefinite matrices}

\begin{lem}
    \label{lem:quadratic_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices, then
    \begin{equation}
        \sum_{j}\lambda_j(A+B)^2 + \sum_{j}\lambda_j(B+C)^2 \leq \sum_{j}\lambda_j(B)^2 + \sum_{j}\lambda_j(A+B+C)^2.
    \end{equation}
\end{lem}
\begin{proof}
    Note that $\sum_{j}\lambda_{j}(X)^2  = \Tr[X^2]$ for any Hermitian operator $X$. Therefore,
    \begin{align}
        \begin{split}
            & \Tr[(B)^2] + \Tr[(A+B+C)^2] - \Tr[(A+B)^2] - \Tr[(B+C)^2] \\
            &\quad= 
            (\Tr[B^2])
            +(\Tr[A^2] + \Tr[B^2] + \Tr[C^2] + 2\Tr[AB] + 2\Tr[BC] + 2 \Tr[AC])\\
            &\qquad\qquad-(\Tr[A^2] + \Tr[B^2] + 2\Tr[AB])
            -(\Tr[B^2] + \Tr[C^2] + 2\Tr[BC])\\
            &\quad= 2 \Tr[AC].
        \end{split}
    \end{align}
    Since $A,C\geq 0$ are both positive semidefinite, we also have $\Tr[AC] \geq 0$ which proves the claim.
\end{proof}

\Citeauthor{bouhtou2008optimization} claim to prove the following family of inequalities \cite[Lemma 5.3]{bouhtou2008optimization}.
\begin{lem}
    \label{lem:bouhtou_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices and let $p \in [0,1]$.
    Then
    \begin{equation}
        \sum_{j}\lambda_j(A+B)^p + \sum_{j}\lambda_j(B+C)^p \geq \sum_{j}\lambda_j(B)^p + \sum_{j}\lambda_j(A+B+C)^p.
    \end{equation}
\end{lem}

The following operator inequality can be used to prove a number of inequalities \cite[Lemma 2.2]{tie2011rearrangement}.
\begin{lem}
    \label{lem:mne_operator_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite matrices and let $k$ be a positive integer.
    Then
    \begin{equation}
        (A+B+C)^{\ot k} + B^{\ot k} \geq (A+B)^{\ot k} + (B+C)^{\ot k}.
    \end{equation}
\end{lem}
\begin{proof}
    The result is obtained from applying a non-commutative multinomial expansion and noting that every summand on the right-hand-side above also appears on the left-hand-side above.
\end{proof}
To obtain an inequality which depends only on the eigenvalues of $B, A+B, B+C$, and $A+B+C$, one can simply pair the positive semidefinite operator in \cref{lem:mne_operator_ineq} with an $\SU(d)$-equivariant and positive semidefinite operator:
\begin{cor}
    \label{cor:isotypic_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite matrices and let $\lambda \vdash k$ be a Young diagram with $k$ boxes.
    Then
    \begin{equation}
        \Tr[P_{\lambda}(A+B+C)^{\ot k}] + \Tr[P_{\lambda}B^{\ot k} \geq \Tr[P_{\lambda}(A+B)^{\ot k}] + \Tr[P_{\lambda}(B+C)^{\ot k}].
    \end{equation}
\end{cor}

\citeauthor{paksoy2014inequalities} proved a family of polynomial inequalities which are generally not functions of just the eigenvalues \cite{paksoy2014inequalities}.
These inequalities are in terms a so-called generalized matrix function first defined by \citeauthor{marcus1965generalized} \cite{marcus1965generalized}.
\begin{defn}[Generalized matrix function]
    Let $H$ be a subgroup of $S_n$ and let $\chi$ be an irreducible character of $H$.
    The \defnsty{generalized matrix function} for the character $\chi$ of $H$ is
    \begin{equation}
        d_{\chi}^{H}(X) = \sum_{\sigma \in H} \chi(\sigma) \prod_{i} X_{i,\sigma(i)}
    \end{equation}
\end{defn}
The notion of a \textit{generalized matrix function} generalizes the notion of an \textit{immanant} (when $H = S_n$) which, in turn, contains as special cases the notion of a permanent and determinant \cite{huber2021matrix}.
\begin{thm}
    \label{thm:ptz_ineqs}
    Let $A,B,C \geq 0$ be positive semidefinite $n\times n$ matrices and let $H$ be a subgroup of $S_n$ with irreducible character $\chi$.
    Then
    \begin{equation}
        d_{\chi}^{H}(A+B+C) + d_{\chi}^{H}(B) \geq d_{\chi}^{H}(A+B) + d_{\chi}^{H}(B+C).
    \end{equation}
\end{thm}
\begin{proof}
    See \cite{paksoy2014inequalities}.
\end{proof}
As a corollary of \cref{thm:ptz_ineqs}, we obtain the following determinant and permanent inequalities.
\begin{cor}
    Let $A,B,C \geq 0$ be positive semidefinite matrices.
    Then
    \begin{align}
        \det(A+B+C) + \det(B) \geq \det(A+B) + \det(B+C), \\
        \mathrm{per}(A+B+C) + \mathrm{per}(B) \geq \mathrm{per}(A+B) + \mathrm{per}(B+C).
    \end{align}
\end{cor}
Note that with the exception of the above determinant inequality, the inequalities provied by \cref{thm:ptz_ineqs} are generally not equivalent to inequalities involving only eigenvalues.

One also has another determinant inequality which is non-linear in determinants \cite[Lemma 2.3]{tie2011rearrangement}.
\begin{lem}
    \label{lem:det_prod_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite matrices.
    Then
    \begin{align}
        \det(A+B+C)\det(B) \leq \det(A+B)\det(B+C).
    \end{align}
\end{lem}
Note that \cref{lem:det_prod_ineq} can be seen as a special of \cref{lem:bouhtou_ineq} in the limit as $p\to 0$; the derivative of \cref{lem:bouhtou_ineq} in $p$ at $p=0$ yields
\begin{equation}
    \label{eq:bouhtou_ineq_p_zero}
    \sum_{j}\log \lambda_j(A+B) + \sum_{j}\log\lambda_j(B+C) \geq \sum_{j}\log\lambda_j(B) + \sum_{j}\log \lambda_j(A+B+C),
\end{equation}
which is equivalent to \cref{lem:det_prod_ineq}.

\begin{thm}
    \label{thm:ssa_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices, then
    \begin{align}
        \label{eq:SSA_like_ineq}
        &\sum_{j}\lambda_j(A+B)\log \lambda_j(A+B)+ \sum_{j}\lambda_j(B+C)\log \lambda_j(B+C) \\
        &\qquad\leq \sum_{j}\lambda_j(B)\log\lambda_j(B) + \sum_{j}\lambda_j(A+B+C)\log\lambda_j(A+B+C).
    \end{align}
\end{thm}

The most comprehensive result concerning the eigenvalues of partial sums of Hermitian matrices we have encountered thus far is \cite[Proposition 2.3]{sagnol2013approximation} which provides the following result.
\begin{thm}
    \label{thm:sagnol_ineq}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices and let $f$ be a real valued and differentiable function over the non-negative real numbers $\mathbb R_{\geq 0}$ such that the derivative of $f$ is an operator monotone function over the positive real numbers $\mathbb R_{>0}$.
    Then the following inequality holds
    \begin{equation}
        \label{eq:sagnol_ineq}
        \Tr f(A+B+C) 
        +
        \Tr f(B)
        \geq
        \Tr f(A+B)
        +
        \Tr f(B+C).
    \end{equation}
\end{thm}
\begin{proof}
    The following proof is a rough summary of the proof found in \cite[Section 2.1]{sagnol2013approximation}.
    Let $X,Y$ be positive semidefinite and consider following difference:
    \begin{equation}
        F(Y|X) \coloneqq \Tr f (X+Y) - \Tr f(Y).
    \end{equation}
    The inequality in \cref{eq:sagnol_ineq} is thus equivalent to $F(C|B) \leq F(A+C|B)$.
    To prove \cref{eq:sagnol_ineq} it suffices to prove that the value of $F(Y|X)$ is a nondecreasing function under $Y \mapsto Y+Z$ for arbitrary positive semidefinite $X,Y$, and $Z$.
    To establish the claim, let $\epsilon > 0$ be small and let $Z = \epsilon W$ for non-zero positive semidefinite $W$.
    It can be shown\footnote{The statement follows, without loss of generality, by assuming that all the matrices involved are positive \textit{definite}. This assumption is permitted because $f$ is assumed continuous and the eigenvalues of a matrix are also continuous functions (see \cite{sagnol2013approximation} for a detailed explanation).} that
    \begin{align}
        (\partial_{Y}F(Y|X))(W)
        &\coloneqq\lim_{\epsilon \to 0} \frac{1}{\epsilon} \left( F(Y+\epsilon W|X) - F(Y|X) \right) \\
        &= \lim_{\epsilon \to 0} \frac{1}{\epsilon} \left( \Tr f(X+Y+\epsilon W) - \Tr f(Y+\epsilon W) - \Tr f(X+Y) + \Tr f(Y) \right) \\
        &= \Tr\left[\left( f'(X+Y)-f'(Y)\right) W\right].
    \end{align}
    Using the assumption that $f'$ is an operator monotone together with $X+Y \geq Y$ (because $X \geq 0$), we have $f'(X+Y) \geq f'(Y)$ and thus $f'(X+Y)-f'(Y) \geq 0$.
    Therefore $(\partial_{Y}F(Y|X))(W) \geq 0$ for all $W$ and thus $F(Y|X)$ is nondecreasing with respect to $Y \mapsto Y+Z$ as desired.
\end{proof}

As corollaries one obtains some of the previously mentioned inequalities for various functions $f$ satisfying the conditions of the theorem; for example, choosing $f(x) = x \log x$ yields \cref{thm:ssa_ineq}, while choosing $f(x) = -x^p$ for $p\in(0,1]$ yields \cref{lem:bouhtou_ineq}, and choosing $f(x) = x^p$ for $p \in (1, 2]$ yields the following.

\begin{cor}
    Let $A,B,C \geq 0$ be positive semidefinite Hermitian matrices and let $p \in (1,2]$.
    Then
    \begin{equation}
        \sum_{j}\lambda_j(A+B)^p + \sum_{j}\lambda_j(B+C)^p \leq \sum_{j}\lambda_j(B)^p + \sum_{j}\lambda_j(A+B+C)^p.
    \end{equation}
\end{cor}

\section{The additive quantum entropy cone}

\subsection{Discovering the direct sum quantum entropy cone}

One of the motivations for studying inequalities between von Neumann entropies of subsytems of multipartite quantum states is because the closure $\bar {\mathcal M_n}$ of the set of possible \textit{allocations of von Neumann entropies},
\begin{equation}
    \mathcal M_{n} \coloneqq \{ (H_{\mathrm{vN}}(\Tr_{[n]\setminus S}(\rho_{12\cdots n})))_{\emptyset \neq S\subseteq [n]} \mid \rho_{12\cdots n} \in \mathcal D(\mathcal H_1 \ot \cdots \ot \mathcal H_n) \},
\end{equation}
was shown by \citeauthor{pippenger2003inequalities} to be a convex cone and thus it suffices to restrict ones attention to linear inequalities of von Neumann entropies \cite{pippenger2003inequalities}.

One might also be interested in set of possible allocations of some other function $f$ of marginals of quantum states, or in our case, the possible allocations of some other function when evaluated over all \textit{partial sums} of a tuple of positive semidefinite Hermitian matrices.
To this end, we consider the following set
\begin{equation}
    \mathcal A_{f,n} \coloneqq \{ \left(f(\sum_{i\in S}X_i)\right)_{\emptyset \neq S\subseteq [n]} \mid X_1, \ldots, X_n \geq 0\}.
\end{equation}
\textbf{Question:} \textit{Does there exist a function $f$ such that $\bar {\mathcal A}_{f,n}$ forms a convex cone?}

Following the argument due to \citeauthor{pippenger2003inequalities}, the closure $\bar {\mathcal A}_{f,n}$ would form a convex cone if an only if $\mathcal M_{f,n}$ was both \textit{additive} and \textit{approximately dilutable}.
Ignoring the condition of approximate dilutability altogether, what would it take to prove that $\mathcal M_{f,n}$ was additive?
A constructive proof would be the following: given two $n$-tuples $X = (X_1, \ldots, X_n)$ and $Y = (Y_1, \ldots, Y_n)$ with allocations $\mathcal M_{f,n}(X)$ and $\mathcal M_{f,n}(Y)$, compute a new $n$-tuple $Z = (Z_1, \ldots, Z_n)$ with allocation $\mathcal M_{f,n}(Z)$ such that
\begin{equation}
    \mathcal M_{f,n}(Z) = \mathcal M_{f,n}(X) +  \mathcal M_{f,n}(Y),
\end{equation}
or in other words,
\begin{equation}
    \label{eq:addivity_partial_sums}
    \forall S \subseteq [n] \quad : \quad 
    f\left(\sum_{i\in S}Z_i\right)
    = 
    f\left(\sum_{i\in S}X_i\right) +
    f\left(\sum_{i\in S}Y_i\right).
\end{equation}
In \cite{pippenger2003inequalities}, \citeauthor{pippenger2003inequalities} used two properties of von Neumann entropies to prove additivity of $\mathcal A_n$: (i) they are additive across tensor products, i.e. $H(\rho \ot \sigma) = H(\rho) + H(\sigma)$, and (ii) they interact well with partial tracing over subsystems.
In our context, suppose a similar argument worked; let $\oslash$ be some binary operation on Hermitian matrices such that $f$ was additive with respect to this operation:
\begin{equation}
    \forall X, Y : f(X \oslash Y) = f(X) + f(Y).
\end{equation}
There are numerous pairs $(f, \oslash)$ of functions and binary operations which satisfy the above equation: e.g., entropies and tensor products $(H, \ot)$, traces and direct sums $(\Tr, \oplus)$, log traces/determinants and tensor products $(\log \Tr, \ot)$, $(\log \det, \ot)$, etc.

A potential difficulty arises when trying to find a function/operation pair which additionally satisfies \cref{eq:addivity_partial_sums}.
If $f$ is additive across some $\oslash$ and \cref{eq:addivity_partial_sums} holds, then we could conclude that $\mathcal M_{f,n}$ was additive. 
Exploring this possibilty further, to prove additivity of $\mathcal M_{f,n}$, it suffices to find an $n$-tuple $Z = (Z_1, \ldots, Z_n)$ satisfying the equations
\begin{equation}
    \label{eq:proposed_constraint}
    \forall S \subseteq [n] \quad : \quad 
    \sum_{k\in S}Z_k
    = 
    \left(\sum_{i\in S}X_i\right) \oslash
    \left(\sum_{j \in S}Y_j\right).
\end{equation}
In other words, the new $n$-tuple $Z = (Z_1, \ldots, Z_n)$ is determined by the binary operation $\oslash$ because $Z_k = X_k \oslash Y_k$.
Therefore, the binary operation $\oslash$ would need to satisfy
\begin{equation}
    \forall S \subseteq [n] \quad : \quad 
    \sum_{k\in S} \left(X_k \oslash Y_k\right)
    = 
    \left(\sum_{i\in S}X_i\right) \oslash
    \left(\sum_{j\in S}Y_j\right).
\end{equation}
This above equation is the ``additive analogue'' of the fact that tensor products and partial traces are compatible meaning
\begin{equation}
    \forall S \subseteq [n] \quad : \quad 
    \Tr_{S} \left(\rho^{X}_{1 2 \cdots n} \ot \rho^{Y}_{1 2 \cdots n}\right)
    = 
    \Tr_{S} \left(\rho^{X}_{1 2 \cdots n}\right) \ot \Tr_{S}\left(\rho^{Y}_{1 2 \cdots n}\right).
\end{equation}
To determine which binary operation $\oslash$ is needed, consider an instance of \cref{eq:proposed_constraint} corresponding to subset $S$ such that $\abs{S} = 2$:
\begin{equation}
    (X_1 \oslash Y_1) + (X_2 \oslash Y_2) = (X_1+X_2)\oslash (Y_1+Y_2).
\end{equation}
From here it is clear that $\oslash$ can be taken to be the direct sum $\oplus$ and the function $f$ any function of Hermitian matrices which is the trace of some function of eigenvalues, i.e. $f = \Tr \circ g$.
In this setting, the desired relation follows:
\begin{equation}
    f(X \oslash Y) = \Tr(g(X \oplus Y)) = \Tr(g(X) \oplus g(Y)) = \Tr(g(X)) + \Tr(g(Y)) = f(X) + f(Y).
\end{equation}

We now return to the question of approximate dilutability.

\textbf{Followup question:} \textit{Does there exist a function $g$ such that $\mathcal M_{\Tr\circ g,n}$ is approximately dilutable?}

In order for $\bar {\mathcal M}_{\Tr\circ g,n}$ to form a convex cone, it must be that ${\mathcal M}_{\Tr\circ g,n}$ is both additive and approximately dilutable.
Fortunately, there exists choices for $g$ such that approximate dilutability holds. 

For example, let $g(x) = -x\log x$.
Take $\mu > 0$ and note that\footnote{The function $g(x) = -x \log x$ applied to positive semidefinite Hermitian matrices satisfies $\Tr(g(\mu X)) = \mu \Tr(g(X)) + g(\mu) \Tr(X)$.}
\begin{align*}
    H(\mu X) 
    &= \Tr(g ( \mu X ))  \\
    &= -\sum_{j} \lambda_{j}(\mu X) \log \lambda_{j}(\mu X) \\
    &= -\mu \sum_{j} \lambda_{j}(X) \log \lambda_{j}(\mu X) \\
    &= -\mu \sum_{j} \lambda_{j}(X) \log \lambda_{j}(X)-\mu \sum_{j} \lambda_{j}(X) \log \mu \\
    &= \mu \Tr(g(X)) - \mu\log \mu  \Tr(X) \\
    &= \mu H(X) - \mu\log \mu  \Tr(X).
\end{align*}
Therefore, for any $C>0$ and $\epsilon > 0$, one can take $\delta > 0$ sufficiently small such that $- \mu \log \mu < C \epsilon$ holds for all $\mu \leq \delta$ implying
\begin{equation}
    \abs{ \Tr(g ( \mu X )) - \mu \Tr(g(X)) } = \abs{- \mu\log \mu  \Tr(X)} \leq C \Tr(X) \epsilon \leq \epsilon,
\end{equation}
where the last inequality follows from choosing $C = \Tr(X)^{-1}$ for all partial sums $X = \sum_{i\in S}X_i$ under consideration.
%
For another example, let $h_{\beta}(x) = \beta x - x \log x$, then
\begin{equation}
    \Tr(h_{\beta}(X)) = \beta \Tr(X) + \Tr(g(X)),
\end{equation}
which implies for $\mu > 0$
\begin{equation}
    \Tr(h_{\beta}(\mu X)) = \beta \mu \Tr(X) + \Tr(g(\mu X)) = \beta \mu \Tr(X) + \mu \Tr(g(X)) - \mu\log \mu  \Tr(X) = \mu \Tr(h_{\beta}(X)) - \mu\log \mu \Tr(X).
\end{equation}
In other words, shifting the function $\Tr \circ g$ by any multiple of the trace also produces a set $\mathcal M_{\Tr \circ h_{\beta}, n}$ which is approximately dilutable (and also additive).
Of course, this is not surprising given that partial sum of matrices always satisfy certain linear constraints (namely $\Tr(X_{S \sqcup T}) = \Tr(X_{S}) + \Tr(X_{T})$ for all disjoint $S,T\subseteq [n]$) and thus line in a linear subspace.

\begin{rem}
    There are two related formulations of the idea of partial sums of positive semidefinite Hermitian matrices.
    The first formulation is the formulation presented in this section; specifically, one is interested in Hermitian matrices $X$ which are positive semidefinite $X \geq 0$ and small in the sense that $\norm{X}_{\infty} \leq 1$ and more generally $\norm{X}_{\infty} < \epsilon$ for arbitrarily small $\epsilon > 0$.
    The second formulation aims to instead normalize all of the non-zero matrices involved by letting $X = \Tr(X) \bar X$ where $\bar X = X/\Tr(X)$.
    The consequences of this normalisation scheme is two-fold.
    First the function $f = \Tr \circ g$ can be reexpressed by setting $\mu = \Tr(X)^{-1}$ to obtain
    \begin{equation}
        \Tr(g (\bar X )) = \frac{\Tr(g(X))}{\Tr(X)} + \log\Tr(X) 
    \end{equation}
    or equivalently
    \begin{equation}
        \Tr(g(X)) = \Tr(X)\Tr(g (\bar X ))- \Tr(X)\log\Tr(X).
    \end{equation}
    The second consequence is that partial sums become 
    \begin{equation}
        \sum_{i\in S}X_i = \sum_{i\in S} \Tr(X_i) \bar X_i
    \end{equation}
    which implies that \textit{normalized} partial sums can be expressed as convex combinations:
    \begin{equation}
        \bar X_{S} = \frac{\sum_{i\in S}X_i}{\Tr\left(\sum_{i\in S}X_i\right)} = \sum_{i\in S} \frac{\Tr(X_i)}{\Tr\left(\sum_{i\in S}X_i\right)} \bar X_i = \sum_{i \in S} p_i \bar X_i
    \end{equation}
    where $p_i = {\Tr(X_i)}/{\Tr\left(\sum_{i\in S}X_i\right)}$.
    The second formulation would then define allocations of entropies of ensembles $\mathcal E = \{(p_i, \rho_i)\}_{i\in[n]}$ of density matrices by setting
    \begin{equation}
        \alpha_{\mathcal E}(S) 
        = H(\sum_{i \in S} p_i \rho_i)
        = p_S H\left(\sum_{i \in S} \frac{p_i}{p_S} \rho_i\right) - p_S \log p_S
    \end{equation}
    where $p_S = \sum_{i\in S} p_i$ and $H(X) = -\sum_{j} \lambda_j(X) \log \lambda_j(X)$ is the von Neumann entropy function (freely extended to consider subnormalized matrices).
\end{rem}

\subsection{The additive entropy cone}

The purpose of this section is to introduce the notion of an ``additive'' quantum entropy cone which serves as a kind of analogy to the ordinary quantum entropy cone considered by \citeauthor{pippenger2003inequalities} \cite{pippenger2003inequalities}.
Formally, quantum states $\rho_{ABC}$ and their reduced states ($\rho_{AB}$, etc.) are replaced by tuples of positive semidefinite operators $(X, Y, Z)$ and their partial sums ($X +Y$, etc.).
Moreover the von Neumann entropy of each reduced state is replaced by an unnormalized entropy of each partial sum.
Using essentially the same approach adopted by \cite{pippenger2003inequalities}, we prove that the topological closure of the set of all such ``allocations'' of unnormalized entropies of partial sums forms a convex cone, which we term the ``additive quantum entropy cone''.
Moreover, we prove inequalities constraining the boundary of the additive quantum entropy cone which parallel the basic positivity, subaddivity and strong subadditivity inequalities which constrain the boundary of the ordinary quantum entropy cone.

\subsubsection{The shifted unnormalized entropy function}

The primary quantity of interest will be a real-valued function of eigenvalues which resembles von Neumann entropies.
\begin{defn}
    Let $\beta \in \mathbb R$ be a constant and define a continuous function $h_{\beta} : \mathbb R_{\geq 0} \to \mathbb R$ by
    \begin{equation}
        h_{\beta} (x) = 
        \begin{cases}
            \beta x - x \log x & x > 0, \\
            0 & x = 0,
        \end{cases}
    \end{equation}
    and let $H_{\beta}$ be the real-valued function on PSD matrices given by
    \begin{equation}
        \label{eq:shifted_unnormalized_entropy}
        H_{\beta}(X) = \Tr(h_{\beta}(X)) = \sum_{i=1}^{d} h_{\beta}(\lambda_i(X)).
    \end{equation}
    The quantity $H_{\beta}(X)$ will be called the \textbf{(shifted) unnormalized entropy} of $X$. 
\end{defn}
\begin{rem}
    For the ease of notation, we will henceforth adopt the convention that $0\log 0 \coloneqq 0$ so that one can write $h_{\beta}(x) = \beta x - x \log x$ without consideration for the $x = 0$ case explicitly.
\end{rem}

\begin{rem}
    It is worth remarking why $H_{\beta}$ is termed the (shifted) unnormalized entropy.
    Suppose $(\lambda_1, \ldots, \lambda_d)$ are the eigenvalues of $X$.
    When $\beta = 0$, we have the following 
    \begin{equation}
        H_{0}(X) = - \sum_{i=1}^{d} \lambda_i \log \lambda_i.
    \end{equation}
    We elect to call $H_0(X)$ the \textit{unnormalized} entropy of $X$ because it coincides with the von Neumann entropy when restricted inputs where $\Tr(X) = 1$ (i.e. when $X$ is a density matrix).
    In general, $H_{0}(X)$ is related to the von Neumann entropy of the \textit{normalization} of $X$ according to the formula
    \begin{equation}
        H_{\mathrm{vN}}\left(\frac{X}{\Tr(X)}\right) = H_{0}\left(\frac{X}{\Tr(X)}\right) = \frac{1}{\Tr(X)} H_0(X) + \log \Tr(X),
    \end{equation}
    or equivalently,
    \begin{equation}
        H_0(X) = \Tr(X)H_{0}\left(\frac{X}{\Tr(X)}\right) - \Tr(X)\log \Tr(X).
    \end{equation}
    The term $H_{\beta}(X)$ is called the \textit{shifted} or $\beta-$\textit{shifted} unnormalized entropy because $H_{\beta}(X) = H_0(X) + \beta \Tr(X)$.
\end{rem}

The following lemma lists some basic properties of the function $H_{\beta}$.
\begin{lem}
    Let $\beta \in \mathbb R$, let $H_{\beta}$ be as defined by \cref{eq:shifted_unnormalized_entropy} and let $X$ be a PSD matrix.
    \begin{itemize}
        \item $H_{\beta}$ is a spectral function: for all unitary matrices $U$,
            \begin{equation}
                H_{\beta}(X) = H_{\beta}(U X U^{\dagger}).
            \end{equation}
        \item If $(\lambda_1, \ldots, \lambda_d)$ are the eigenvalues of $X$, then
            \begin{align}
                \begin{split}
                    H_{\beta}(X) 
                    &= \sum_{i=1}^{d} \left(\beta \lambda_i - \lambda_i \log \lambda_i\right).
                \end{split}
            \end{align}
        \item Let $0_{d}$ be the $d\times d$ zero matrix. Then
            \begin{equation}
                H_{\beta}(0_{d}) = 0
            \end{equation}
        \item Let $I_{d}$ be the $d\times d$ identity matrix. Then
            \begin{equation}
                H_{\beta}(I_{d}) = \beta d
            \end{equation}
    \end{itemize}
\end{lem}

It is important to note that $H_{\beta}(X)$ is not a homogeneous function in $X$ meaning in general
\begin{equation}
    H_{\beta}(X) \neq H_{\beta}(\gamma X).
\end{equation}
Nevertheless, there is a sense it which it is homogeneous up to a modification to the value of $\beta$ which depends on the scaling parameter $\gamma$.
\begin{lem}
    \label{lem:deformation_homogeneous}
    Let $\gamma > 0$ be non-zero. Then $H_{\beta}$ is deformation-homogeneous in the sense that
    \begin{equation}
        \label{eq:deformation_homogeneous}
        H_{\beta}(\gamma X) = \beta \gamma \Tr(X) + \gamma H_{0}(X) - \gamma \log \gamma \Tr(X) = \gamma H_{\beta - \log \gamma}(X).
    \end{equation}
    As a special case, letting $\beta = \log \gamma$ produces
        \begin{equation}
            H_{0}(X) = \frac{1}{\gamma} H_{\log \gamma}(\gamma X),
        \end{equation}
    and as a further special case, letting $\gamma = \Tr(X)^{-1}$ yields
        \begin{equation}
            H_{0}(X) = \Tr(X) H_{-\log \Tr(X)}\left(\frac{X}{\Tr(X)}\right).
        \end{equation}
\end{lem}

\subsubsection{Inequalities for the unnormalized entropy function}

The first inequality is the most basis.
It identifies that for sufficiently small PSD matrices $X$, $H_{\beta} (X)$ is non-negative.
\begin{lem}
    Let $X$ be a PSD matrix with sufficiently small maximum eigenvalue such that
    \begin{equation}
        \label{eq:norm_beta_cond}
        \norm{X}_{\infty} \leq \exp (\beta).
    \end{equation}
    Then,
    \begin{equation}
        H_{\beta}(X) \geq 0,
    \end{equation}
    with equality (for $X$ satisfying \cref{eq:norm_beta_cond}) holding only if $X$ has eigenvalues in the set $\{0, \exp(\beta)\}$.
\end{lem}
\begin{proof}
    If $(\lambda_1, \ldots, \lambda_d)$ are the eigenvalues of $X$, then $H_{\beta}(X)$ equals
    \begin{equation}
        \label{eq:eigen_sum_Hbeta}
        H_{\beta}(X) = \sum_{i=1}^{d} \lambda_i (\beta - \log \lambda_i).
    \end{equation}
    Under the conditions that $X \geq 0$ is PSD and satisfying \cref{eq:norm_beta_cond}, we have
    \begin{equation}
        \lambda_i \geq 0, \qquad \text{and} \qquad \beta - \log \lambda_i \geq 0,
    \end{equation}
    respectively.
    Therefore each summand in \cref{eq:eigen_sum_Hbeta} is positive and thus $H_{\beta}(X) \geq 0$ holds.
    To prove the claim regarding equality $H_{\beta}(X)$ we note that each summand in \cref{eq:eigen_sum_Hbeta} must be zero and thus either $\lambda_i = 0$ or $\lambda_i = \exp(\beta)$.
\end{proof}

    \textcolor{blue}{TC: If is unclear at the moment whether it is possible for $X$ to satisfy $H_{\beta}(X) = 0$ while having eigenvalues different from $\{0,\exp(\beta)\}$ if one allows for $\norm{X}_{\infty} > \exp(\beta)$.}
\end{rem}

\begin{thm}
    \label{thm:unnormalized_entropy_ssa}
    Let $\beta \in \mathbb R$ be arbitary and let $X, Y, Z \geq 0$ be PSD matrices. 
    Then $H_{\beta}$ is both subadditive
    \begin{equation}
        \label{eq:subadditive_Hbeta}
        H_{\beta}(X+Y) \leq H_{\beta}(X) + H_{\beta}(Y),
    \end{equation}
    and also strongly subadditive in the sense that
    \begin{equation}
        \label{eq:strong_subadditive_Hbeta}
        H_{\beta}(X+Y+Z) + H_{\beta}(Y) \leq H_{\beta}(X+Y) + H_{\beta}(Y+Z).
    \end{equation}
\end{thm}
\begin{proof}
    We begin by noting that subadditivity (\cref{eq:subadditive_Hbeta}) is a special case of strong subadditivty (\cref{eq:strong_subadditive_Hbeta}) under the substitution $(X,Y,Z) \mapsto (X, 0, Y)$ because $H_{\beta}(0) = 0$ for any value of $\beta$.
    Therefore it suffices to just prove strong subadditivity (\cref{eq:strong_subadditive_Hbeta}).
    Furthermore, it sufficies to prove just the $\beta = 0$ case of \cref{eq:strong_subadditive_Hbeta} because $H_{\beta}(X) = \beta \Tr(X) + H_0(X)$ and 
    \begin{equation}
        \label{eq:balanced}
        \Tr(X+Y+Z) + \Tr(Y) = \Tr(X+Y) + \Tr(Y+Z).
    \end{equation}
    In any case, strong subaddivity of $H_{\beta}$ or arbitary $\beta$ holds as an special case of \cref{thm:sagnol_ineq} where the function $f$ is taken to be the function 
    \begin{equation}
        f(x) \coloneqq -h_{\beta}(x) = -\beta x + x \log x.
    \end{equation}
    The derivative of $-h_{\beta}$ is
    \begin{equation}
        (-h_{\beta})'(x) = 1-\beta + \log x
    \end{equation}
    which is known to be operator monotone \cite{chansangiam2013operator} on $\mathbb R_>0$ as required.
\end{proof}

\subsubsection{Comparison to concavity of von Neumann entropy}
At this stage it is worth comparing how the inequalities in \cref{thm:unnormalized_entropy_ssa} compare to known inequalities concerning von Neumann entropy.
Suppose $\rho_X$ and $\rho_Y$ are two density matrices and $\lambda \in [0,1]$ is a convex parameter.
Then it is known that the von Neumann entropy satisfies \textit{concavity}:
\begin{equation}
    \lambda H_{\mathrm{vN}}(\rho_X) + (1-\lambda) H_{\mathrm{vN}}(\rho_X) \leq  H_{\mathrm{vN}}(\lambda\rho_X + (1-\lambda) \rho_Y),
\end{equation}
but also an associated upper bound:
\begin{equation}
    H_{\mathrm{vN}}(\lambda\rho_X + (1-\lambda) \rho_Y) \leq \lambda H_{\mathrm{vN}}(\rho_X) + (1-\lambda) H_{\mathrm{vN}}(\rho_X) + h_{\mathrm{bin}}(\lambda)
\end{equation}
where $h_{\mathrm{bin}}(\lambda)$ is the binary entropy function:
\begin{equation}
    h_{\mathrm{bin}}(\lambda) = -\lambda \log \lambda - (1-\lambda) \log \lambda.
\end{equation}

\textbf{(Upper bound):}
We first show that this upper bound is related to subadditvity of $H_0$.
If we let $p_X, p_Y \geq 0$ be weights such that
\begin{equation}
    \lambda = \frac{p_{X}}{p_{X} + p_{Y}}, \qquad 1-\lambda = \frac{p_{Y}}{p_{X} + p_{Y}},
\end{equation}
then the upper bound above can be equivalently written in terms of the unnormalized entropy $H_0$ of
\begin{equation}
    \sigma_X \coloneqq p_X \rho_X, \qquad \sigma_Y = p_Y \rho_Y,
\end{equation}
as follows 
\begin{equation}
    H_{0}(\sigma_X + \sigma_Y) \leq H_0(\sigma_X) + H_0(\sigma_Y).
\end{equation}
We note that this expression is equivalent to the subadditivity of $H_0$ (\cref{eq:subadditive_Hbeta} from \cref{thm:unnormalized_entropy_ssa}).
This is because every non-zero PSD matrix $\sigma_X$ can be equivalently expressed in terms of a positive weight $p_X$ and a density matrix $\rho_X$ according to the formula:
\begin{equation}
    (p_X, \rho_X) \coloneqq \left(\Tr(\sigma_X), \frac{\sigma_X}{\Tr(\sigma_X)}\right).
\end{equation}

\textbf{(Lower bound/concavity):}
It is also possible to express the concavity of von Neumann entropy as a kind of superadditivity of a different function, namely the function 
\begin{equation}
    g(X) = \Tr(X) H_{\mathrm{vN}}\left(\frac{X}{\Tr(X)}\right) = H_0(X) + \Tr(X) \log \Tr(X).
\end{equation}
In this notation, the concavity of von Neumann entropy is equivalent to
\begin{equation}
    g(\sigma_X + \sigma_Y) \geq g(\sigma_X) + g(\sigma_Y).
\end{equation}

\subsubsection{The additive unnormalized entropy cone}

We are now in a position to actually define allocations of unnormalized entropies of partial sums.
\begin{defn}
    Let $\boldsymbol{X} \coloneqq (X_1, \ldots, X_n)$ be an $n$-tuple of positive semidefinite Hermitian matrices.
    Define the \textbf{allocation of unnormalized entropies} of $\boldsymbol{X}$ to be the element of $\mathbb R^{2^n}$ given by
    \begin{equation}
        \alpha_{\beta, n}(\boldsymbol{X}) = \left( H_{\beta}(\sum_{i\in S}X_i)\right)_{S \subset [n]}.
    \end{equation}
    The set of all possible allocations of unnormalized entropies will be denoted by $\mathcal A_{\beta, n}$.
\end{defn}
\begin{exam}
    For example, when $n=3$, the allocation of unnormalized entropies for $\boldsymbol{X} = (X,Y,Z)$ is
    \begin{equation}
        \alpha_{\beta, 3}(X, Y, Z) = \left( 0, H_{\beta}(X), H_{\beta}(Y), H_{\beta}(Z), H_{\beta}(Y+Z), H_{\beta}(X+Z), H_{\beta}(X+Y), H_{\beta}(X+Y+Z) \right).
    \end{equation}
\end{exam}

Our goal will be to prove that the closure, $\bar {\mathcal A}_{\beta, n}$, of the set of possible allocations of unnormalized entropies is a convex cone.
Following \cite{pippenger2003inequalities}, the closure $\bar {\mathcal A}_{\beta, n}$ is a convex cone if and only if $\mathcal A_{\beta, n}$ is both \textit{additive} and \textit{approximately dilutable}.
To begin we show that $\mathcal A_{\beta, n}$ additive (closed under addition).

\begin{lem}
    \label{lem:additive}
    If $\boldsymbol{X}=(X_1, \ldots, X_n)$ and $\boldsymbol{X'}=(X'_1, \ldots, X'_n)$ are $n$-tuples of PSD matrices with allocations $\alpha_{\beta, n}(\boldsymbol{X})$ and $\alpha_{\beta, n}(\boldsymbol{X'})$,
    then the $n$-tuple
    \begin{equation}
        \boldsymbol{X}\oplus \boldsymbol{X'} \coloneqq (X_1 \oplus X'_1, \ldots, X_n \oplus X'_n),
    \end{equation}
    has allocation given by
    \begin{equation}
        \alpha_{\beta, n}(\boldsymbol{X}\oplus\boldsymbol{X'}) = \alpha_{\beta, n}(\boldsymbol{X}) + \alpha_{\beta, n}(\boldsymbol{X'}).
    \end{equation}
    In other words, the set $\mathcal A_{\beta, n}$ is closed under addition.
\end{lem}
\begin{proof}
    This follows directly from the fact that $H_{\beta}$ is additive across direct sums,
    \begin{equation}
        H_{\beta}(X \oplus X') = H_{\beta}(X) + H_{\beta}(X'),
    \end{equation}
    and the fact that the direct sum is distributive across sums so that
    \begin{equation}
        \sum_{i\in S} X_i \oplus X'_i = \left(\sum_{j\in S} X_j\right) \oplus \left(\sum_{k\in S} X'_k\right).
    \end{equation}
\end{proof}

Next we show that $\mathcal A_{\beta, n}$ is approximately dilutable.
\begin{lem}
    \label{lem:approx_dilute}
    The $\mathcal A_{\beta, n}$ is approximately dilutable.
\end{lem}
\begin{proof}
    Take $\mu > 0$ and consider the allocation of $X \geq 0$ compared to the allocation of $\mu X \geq 0$.
    By
    \begin{align}
        H_{\beta}(\mu X) 
        &= \beta \Tr(\mu X) + H_{0}(\mu X), \\
        &= \beta \mu \Tr(X) + \mu H_{0}(X) - \Tr(X) \mu \log \mu, \\
        &= \mu H_{\beta}(X) - \mu\log \mu  \Tr(X).
    \end{align}
    Therefore, for any constant $C>0$ and $\epsilon > 0$, one can take $\delta > 0$ sufficiently small such that $- \mu \log \mu < C \epsilon$ holds for all $\mu \leq \delta$ implying
    \begin{equation}
        \abs{ H_{\beta}(\mu X) - \mu H_{\beta}(X) } = \abs{- \mu\log \mu  \Tr(X)} \leq C\epsilon \Tr(X) .
    \end{equation}
    Therefore, the distance between the allocations of $\mu \boldsymbol{X}$ and $\boldsymbol{X}$ is bounded by
    \begin{equation}
        \norm{\alpha_{\beta,n}(\mu \boldsymbol{X}) - \mu \alpha_{\beta,n}(\boldsymbol{X})}_{\infty} \leq C \epsilon \Tr\left(\sum_{i=1}^{n} X_i \right).
    \end{equation}
    It one chooses $C = \Tr(X_1 + \cdots + X_n)^{-1}$, then one obtains for all $\epsilon > 0$ the following holds for all sufficiently small $\mu$,
    \begin{equation}
        \norm{\alpha_{\beta,n}(\mu \boldsymbol{X}) - \mu \alpha_{\beta,n}(\boldsymbol{X})}_{\infty} \leq \epsilon,
    \end{equation}
    and therefore the claim holds.
\end{proof}

\begin{thm}
    The closure, $\bar {\mathcal A}_{\beta, n}$, of the set of all possible allocations of (shifted) unnormalized entropies is a convex cone.
\end{thm}
\begin{proof}
    This follows \cite[Theorem 2.2]{pippenger2003inequalities} and \cref{lem:additive,lem:approx_dilute}.
\end{proof}

It appears that the structure of the cones $\bar {\mathcal A}_{\beta, n}$ do not depend on the value of $\beta$ in any meaningful way and they are all related by an appropriate re-scaling.
\begin{cor}
    Let $\beta_1, \beta_2 \in \mathbb R$.
    There exists a bijection from $\mathcal A_{\beta_1, n}$ to $\mathcal A_{\beta_2, n}$ given by multiplying componentwise by a factor of $\gamma = \exp(\beta_1 - \beta_2)$.
\end{cor}
\begin{proof}
    The deformation-homogeneous property of $H_{\beta}$ (\cref{eq:deformation_homogeneous}) implies
    that for $\gamma \coloneqq \exp(\beta_1 - \beta_2)$, the following holds
    \begin{equation}
        H_{\beta_2}(\gamma X) = \gamma H_{\beta_1}(X).
    \end{equation}
    By multiplying a given $n$-tuple $\boldsymbol{X} = (X_1, \ldots, X_n)$ entry-wise by the same factor $\gamma$, one concludes that if $\alpha_{\beta_1, n} \in \mathcal A_{\beta_1, n}$ then $\exp(\beta_1 - \beta_2) \alpha_{\beta_1, n} \in \mathcal A_{\beta_2, n}$.
\end{proof}

Given that each of the convex cones associated to allocations of unnormalized entropies are essentially the same, we will henceforth refer to the \textbf{additive quantum entropy cone} as the closure of the set associated to $\beta = 0$:
\begin{equation}
    \bar {\mathcal A}_{n} \coloneqq \bar {\mathcal A}_{0, n}.
\end{equation}

\subsubsection{Connection to the multiplicative quantum entropy cone}

The purpose of this section is to compare the \textit{additive} quantum entropy cone, $\bar {\mathcal A}_{0,n}$, to the \textit{(multiplicative)} quantum entropy cone, $\bar {\mathcal M}_{n}$, previously studied by \citeauthor{pippenger2003inequalities} \cite{pippenger2003inequalities}.
Our goal will be to prove how the additive quantum entropy cone, $\bar {\mathcal A}_{0,n}$, lies inside the multiplicative quantum entropy cone, $\bar {\mathcal M}_{n}$, by following a particular family of quantum states.

\begin{defn}
    Let $n, d$ be positive integers and consider the complex vector space $(\mathbb C^{d+1})^{\ot n}$.
    For each $p \in [n]$, let $\{ \ket {0}_p, \ket {1}_p, \ldots, \ket {d}_p \}$ be a basis for $\mathbb C^{d+1}$ (noting that $\ket{i}_{p}$ need not equal $\ket{i}_{q}$ for $p \neq q$).
    For each $k \in \{0, 1, \ldots, n\}$, we define the \textbf{weight-$\boldsymbol{k}$ subspace}, $W_k \subset (\mathbb C^{d+1})^{\ot n}$, to be the span of those vectors in $(\mathbb C^{d+1})^{\ot n}$ of the form 
    \begin{equation}
        \ket{i_{1}}_{1} \ot \cdots \ot \ket{i_{n}}_{n}
    \end{equation}
    such that exactly $k$ elements of the tuple $(i_{1}, \ldots, i_{p}) \in [d+1]^{n}$ are non-zero.
\end{defn}
\begin{rem}
    For example, the weight zero subspace, $W_0$, is one-dimension and is spanned by the vector
    \begin{equation}
        \label{eq:weight_zero}
        \kett{\emptyset} = \ket{0}_{1}\ot \cdots \ket{0}_{n}.
    \end{equation}
    The weight one subspace, $W_1$, is $dn$-dimensional and is spanned by the vectors $\kett{i_p}$ where 
    \begin{equation}
        \label{eq:weight_one}
        \kett{i_p} = \ket{0}_1 \ot \cdots \ot \ket{0}_{p-1} \ot \ket{i} \ot \ket{0}_{p+1} \ot \ket{0}_{n}.
    \end{equation}
    and where $i \in [d]$, $p \in [n]$.
    The weight two subspace, $W_2$, is $d^2 \binom{n}{2}$-dimensional and is spanned by vectors $\kett{i_p, j_q}$ where
    \begin{equation}
        \label{eq:weight_two}
        \kett{i_p, j_q} = \ket{0}_1 \ot \cdots \ot \ket{0}_{p-1} \ot \ket{i} \ot \ket{0}_{p+1} \ot \cdots \ot \ket{0}_{q-1} \ot \ket{j}_{q} \ot \ket{0}_{q+1} \ot \cdots \ot \ket{0}_{n}.
    \end{equation}
    and where $1 \leq p < q \leq d$ and $i,j \in [d]$.
\end{rem}
\begin{rem}
    While the notion of a \textit{weight-$k$} subspace is technically basis dependent, it is only partially basis dependent.
    In particular, it is invariant under independent local changes of basis which fix the zero-element, $\ket{0}_p$ of each party $p$.
    In other words, the weight $k$-subspace is invariant under the group $G = (U(1) \times U(d))^{\times n}$
    where $U(1) \times U(d) \subset U(d+1)$ refers to the subgroup of the unitary group which preserves the subspace spanned by $\ket{0}_p$.
    In particular, a weight space decomposition of $(\mathbb C^{d+1})^{\ot n}$ depends only on a fixed product vector $\ket{\emptyset} \coloneqq \ket{\phi}_1 \ot \cdots \ket{\phi}_n$.
\end{rem}

\begin{defn}
    \label{defn:spec_embed}
    Let $(X_1, \ldots, X_n)$ be a $n$-tuple of $d\times d$ PSD operators and let $(\{\ket i_p\}_{i=0}^{d})_{p\in[n]}$ be an $n$-tuple of bases for $\mathbb C^{d+1}$.
    Let $\Gamma(X_1, \ldots, X_n)$ be the Hermitian operator on $(\mathbb C^{d+1})^{\ot n}$ which is block-diagonal with respect to the weight decomposition of $(\mathbb C^{d+1})^{\ot n}$ with non-zero entries given by
    \begin{equation}
        \label{eq:weight_zero_portion}
        \braa{\emptyset}\Gamma(X_1, \ldots, X_n)\kett{\emptyset} = 1 - \sum_{i=1}^{n} \Tr(X_n),
    \end{equation}
    and
    \begin{equation}
        \label{eq:weight_one_portion}
        \braa{i_p}\Gamma(X_1, \ldots, X_n)\kett{j_q} = \bra{i}_pX_p^{\frac{1}{2}} X_{q}^{\frac{1}{2}} \ket{j}_q.
    \end{equation}
    For example, the coefficients of $\Gamma_{[n]}$ in the weight-two subspace are zero,
    \begin{equation}
        \braa{i_p, i'_{p'}}\Gamma(X_1, \ldots, X_n)\kett{j_q, j'_{q'}} = 0,
    \end{equation}
    as well as all higher weight spaces. 
\end{defn}
\begin{exam}
    Given two PSD $d\times d$ matrices, $(X_1, X_2)$, the matrix $\Gamma_{[2]} \coloneqq \Gamma(X_1, X_2)$ has the following coefficients for all $i,j,k,l \in \{1, \ldots, d\}$.
    \begin{align}
        \braket{00|\Gamma|00} &= 1-\Tr(X_{1})-\Tr(X_{2}), \\
        \braket{i0|\Gamma|j0} &= \braket{i|X_{1}|j}, \\
        \braket{0i|\Gamma|0j} &= \braket{i|X_{2}|j}, \\
        \braket{i0|\Gamma|0j} &= \braket{i|X_{1}^{\frac{1}{2}}X_{2}^{\frac{1}{2}}|j}, \\
        \braket{0i|\Gamma|j0} &= \braket{i|X_{2}^{\frac{1}{2}}X_{1}^{\frac{1}{2}}|j}, \\
        \braket{ik|\Gamma|jl} &= 0.
    \end{align}
\end{exam}
\begin{exam}
    Given three $d\times d$ PSD matrices $X_1, X_2, X_3 \geq 0$, the operator $\Gamma_{[3]} \coloneqq \Gamma(X_1, X_2, X_3)$ is the $(d+1)^{3}\times (d+1)^{3}$ matrix with non-zero coefficients given for all $i,j \in \{1, \ldots, d\}$ by the formulas:
    \begin{align}
        \begin{split}
            \braket{000|\Gamma|000} &= 1-\Tr(X_1+X_2+X_3), \\
            \braket{i00|\Gamma|j00} &= \braket{i|X_1|j}, \\
            \braket{0i0|\Gamma|0j0} &= \braket{i|X_2|j}, \\
            \braket{00i|\Gamma|00j} &= \braket{i|X_3|j}, \\
            \braket{i00|\Gamma|0j0} &= \braket{i|X_1^{\frac{1}{2}}X_2^{\frac{1}{2}}|j}, \\
            \braket{0i0|\Gamma|00j} &= \braket{i|X_2^{\frac{1}{2}}X_3^{\frac{1}{2}}|j}, \\
            \braket{i00|\Gamma|00j} &= \braket{i|X_1^{\frac{1}{2}}X_3^{\frac{1}{2}}|j}, \\
            \braket{0i0|\Gamma|j00} &= \braket{i|X_2^{\frac{1}{2}}X_1^{\frac{1}{2}}|j}, \\
            \braket{00i|\Gamma|j00} &= \braket{i|X_3^{\frac{1}{2}}X_1^{\frac{1}{2}}|j}, \\
            \braket{00i|\Gamma|0j0} &= \braket{i|X_3^{\frac{1}{2}}X_2^{\frac{1}{2}}|j}.
        \end{split}
    \end{align}
\end{exam}

\begin{lem}
    \label{lem:sum_spec_embed}
    Let $\Gamma_{[n]} \coloneqq \Gamma(X_1, \ldots, X_n)$ be defined as in \cref{defn:spec_embed}.
    Then the eigenvalues of $\Gamma_{[n]}$ are equal to the eigenvalues of the sum, 
    \begin{equation}
        X_{[n]} \coloneqq \sum_{i \in 1}^{n}X_i,
    \end{equation}
    plus the additional value of $1-\Tr(X_{[n]})$:
    \begin{equation}
        \mathrm{spec}(\Gamma_{[n]}) \simeq \mathrm{spec}(X_{[n]}) \oplus \{1-\Tr(X_{[n]})\}.
    \end{equation}
\end{lem}
\begin{proof}
    By definition, $\Gamma_{[n]}$ is block diagonal with respect to the weight decomposition.
    As the weight-zero subspace is one-dimensional and spanned by $\kett{\emptyset} = \ket{0}\cdots \ket{0}$, we see that $\kett{\emptyset}$ is an eigenvector of $\Gamma_{[n]}$ with eigenvalue given by $\braa{\emptyset}\Gamma_{[n]}\kett{\emptyset}$ which is given by \cref{eq:weight_one_portion}, i.e.,
    \begin{equation}
        \Gamma_{[n]}\kett{\emptyset} = (1-\Tr(X_{[n]}) \kett{\emptyset}.
    \end{equation}
    For the remaining eigenvalues of $\Gamma_{[n]}$, let $\Phi_{[n]}$ be the operator
    $\Gamma_{[n]}$ with its weight-zero entry set to zero: 
    \begin{equation}
        \Phi_{[n]} \coloneqq \Gamma_{[n]} - (1-\Tr(X_{[n]}) \kett{\emptyset}\braa{\emptyset}.
    \end{equation}
    To prove the claim, it therefore suffices to show that the eigenvalues of $\Phi_{[n]}$ are identical to the eigenvalues of the sum $X_{[n]}$ (with multiplicitly).
    Equivalently, it suffices to prove that the characteristic polynomials of $\Phi_{[n]}$ and $X_{[n]}$ have the same coefficients which is implied by proving that for any positive integer $\ell$, $\Tr(\Phi_{[n]}^{\ell}) = \Tr(X_{[n]}^{\ell})$.
    For $\ell = 1$, the proof follows from performing a single sum over the weight-one subspace
    \begin{equation}
        \Tr(\Phi_{[n]}) = \sum_{p=1}^{n}\sum_{i=1}^{d} \braa{i_p} \Phi_{[n]} \kett{i_p} = \sum_{p=1}^{n}\sum_{i=1}^{d} \bra{i}_pX_{p}\ket{i}_p = \sum_{p = 1}^{n} \Tr(X_{p}) = \Tr(X_{[n]}).
    \end{equation}
    For $\ell > 1$, the proof follows from an $\ell$-fold sum over the weight-one subspace,
    \begin{align}
        \Tr(\Phi_{[n]}^{\ell}) 
        &= \sum_{p_1=1}^{n}\sum_{i_1=1}^{d} \cdots \sum_{p_\ell=1}^{n}\sum_{i_\ell=1}^{d} 
        \braa{(i_1)_{p_1}} \Phi_{[n]} \kett{(i_2)_{p_2}}\cdots \braa{(i_\ell)_{p_\ell}} \Phi_{[n]} \kett{(i_1)_{p_1}}, \\
        &= \sum_{p_1=1}^{n}\sum_{i_1=1}^{d} \cdots \sum_{p_\ell=1}^{n}\sum_{i_\ell=1}^{d} 
        \bra{i_1}_{p_1} X_{p_1}^{\frac{1}{2}}X_{p_2}^{\frac{1}{2}} \ket{i_2}_{p_2}\cdots \bra{i_\ell}_{p_\ell} X_{p_\ell}^{\frac{1}{2}} X_{p_1}^{\frac{1}{2}}\ket{i_1}_{p_1}, \\
        &= \sum_{p_1=1}^{n}\cdots \sum_{p_\ell=1}^{n} 
        \Tr[X_{p_1}^{\frac{1}{2}}X_{p_2}^{\frac{1}{2}}\cdots X_{p_\ell}^{\frac{1}{2}} X_{p_1}^{\frac{1}{2}}], \\
        &= \sum_{p_1=1}^{n}\cdots \sum_{p_\ell=1}^{n} 
        \Tr[X_{p_1}X_{p_2}\cdots X_{p_\ell}], \\
        &= \Tr\left[\left(\sum_{p_1=1}^{n}X_{p_1}\right)\cdots \left(\sum_{p_\ell=1}^{n}X_{p_\ell}\right)\right], \\
        &= \Tr(X_{[n]}^{\ell}).
    \end{align}
    Therefore $X_{[n]}$ and $\Phi_{[n]}$ have the same eigenvalues with multiplicity which proves the claim.
\end{proof}
\begin{cor}
    Let everything be as in \cref{lem:sum_spec_embed}.
    Then $\Tr(\Gamma_{[n]}) = 1$.
    If the eigenvalues of the sum $X_{[n]}$ satisfy
    \begin{equation}
        \Tr(X_{[n]}) \leq 1,
    \end{equation}
    then $\Gamma_{[n]}$ is positive semidefinite, and thus $\Gamma_{[n]}$ is a density matrix.
    In such cases, the von Neumann entropy of $\Gamma_{[n]}$ is related to unnormalized entropy of $X_{[n]}$ by
    \begin{align}
        H_{\mathrm{vN}}\left(\Gamma_{[n]}\right) 
        &= (1-\Tr(X_{[n]}))\log(1-\Tr(X_{[n]})) + H_0(X_{[n]}), \\
        &= h_{\mathrm{bin}}(\Tr(X_{[n]})) + \Tr(X_{[n]})H_0\left(\frac{X_{[n]}}{\Tr(X_{[n]})}\right).
    \end{align}
    Moreover, if the eigenvalues of $X_{[n]}$ satisfy the stronger condition
    \begin{equation}
        \lambda_{1}(X_{[n]}) + \Tr(X_{[n]}) \leq 1,
    \end{equation}
    then the maximum eigenvalue of $\Gamma_{[n]}$ is $1-\Tr(X_{[n]})$.
\end{cor}

\begin{lem}
    \label{lem:partial_sum_embed}
    Let $\Gamma_{[n]} \coloneqq \Gamma(X_1, \ldots, X_n)$ be defined as in \cref{defn:spec_embed}.
    Let $S \subseteq \{1, \ldots, n\}$ be any subset and let $\Gamma_{S} \coloneqq \Gamma( (X_i)_{i \in S})$ with $\Gamma_{\emptyset} \coloneqq 1$.
    Then the partial trace of $\Gamma_{[n]}$ over $[n]\setminus S$ is equal to $\Gamma_{S}$:
    \begin{equation}
        \Tr_{[n]\setminus S} (\Gamma_{[n]}) = \Gamma_{S}.
    \end{equation}
    In particular, letting $S = \{p\}$ yields 
    \begin{equation}
        \Gamma_{p} \coloneqq \Gamma(X_p) = (1-\Tr(X_p)) \oplus X_p = \Tr_{[n] \setminus \{p\}}(\Gamma_{[n]}).
    \end{equation}
\end{lem}
\begin{proof}
    To begin, notice that for any $n$-partite weight-block-diagonal operator $X$ on $(\mathbb C^{d})^{\ot n}$, its partial trace over $S \subseteq [n]$, $\Tr_{[n]\setminus S} (X)$, will also be weight-block-diagonal as an $\card{S}$-partite operator because the weight-block-off-diagonal entries of $\Tr_{[n]\setminus S} (X)$ are sums of weight-block-off-diagonal entries of $X$.
    Therefore, to prove the claim, it suffices to restrict focus to the weight-block-diagonal entries of $\Tr_{[n]\setminus S} (\Gamma_{[n]})$.

    Let $\omega : S \to \{1, \ldots, \card{S}\}$ be the map which assigns to each $p \in S$ its position in $S$ ordered relative to the ordering on $[n]$.
    Then the coefficients of $\Gamma_{[n]}$ in the weight-one subspace and the coefficients of $\Gamma_{S}$ in the weight-one subspace are related for all $p,q \in S$ according to,
    \begin{equation}
        \braa{i_p} \Gamma_{[n]} \kett{j_p} = \bra{i}X_{p}^{\frac{1}{2}} X_{q}^{\frac{1}{2}} \ket{j} = \braa{i_{\omega(p)}} \Gamma_{S} \kett{j_{\omega(q)}}
        \label{eq:marginal_joint_correspondence}
    \end{equation}
    Therefore, the coefficients of $\Tr_{[n]\setminus S}[\Gamma_{[n]}]$ in the weight-one block for all $p,q\in S$ and $i,j \in [d]$ are given by
    \begin{align}
        \braa{i_{\omega(p)}}\Tr_{[n]\setminus S}[\Gamma_{[n]}] \kett{j_{\omega(q)}}
        = \braa{i_{p}}\Gamma_{[n]}\kett{j_{q}}
        = \braa{i_{\omega(p)}} \Gamma_{S} \kett{j_{\omega(q)}}.
    \end{align}
    Moreover, using \cref{eq:marginal_joint_correspondence} again shows that the coefficient of $\Tr_{[n]\setminus S}[\Gamma_{[n]}]$ in the weight-zero block equals
    \begin{align}
        \braa{\emptyset}\Tr_{[n]\setminus S}[\Gamma_{[n]}] \kett{\emptyset}
        &= \braa{\emptyset}\Gamma_{[n]}\kett{\emptyset} + \sum_{p \not \in S} \sum_{i=1}^{d} \braa{i_p} \Gamma_{[n]} \kett{i_p} \\
        &= \braa{\emptyset}\Gamma_{[n]}\kett{\emptyset} + \sum_{p = 1}^{n} \sum_{i=1}^{d} \braa{i_p} \Gamma_{[n]} \kett{i_p} - \sum_{p \in S} \sum_{i=1}^{d} \braa{i_p} \Gamma_{[n]} \kett{i_p} \\
        &= \Tr(\Gamma_{[n]}) - \sum_{p \in S} \sum_{i=1}^{d} \braa{i_{\omega(p)}} \Gamma_{S} \kett{i_{\omega(p)}} \\
        &= \Tr(\Gamma_{[n]}) - (\Tr(\Gamma_{S}) - \braa{\emptyset}\Gamma_{S}\kett{\emptyset}) \\
        &= \braa{\emptyset}\Gamma_{S}\kett{\emptyset}
    \end{align}
    because $\Tr(\Gamma_{[n]}) = \Tr(\Gamma_{S}) = 1$.
    Therefore, $\Gamma_{S} = \Tr_{[n]\setminus S}[\Gamma_{[n]}]$ which proves the claim.
\end{proof}

It is worth emphasizing that the operators, $\Gamma_{[n]} = \Gamma(X_1, \ldots, X_n)$, obtained from an $n$-tuple $(X_1, \ldots, X_n)$ according to \cref{defn:spec_embed} do not exhaust all $n$-partite operators.
Nevertheless, there is an elegant characterization of those $n$-partite operators which can be obtained in this way.
This characterization generalizes \cite[Proposition 11]{christandl2018recoupling}.
\begin{lem}
    Let $\rho_{[n]}$ be an $n$-partite density matrix on $(\mathbb C^{d+1})^{\ot n}$ with rank at most $d+1$ such that there exists a product eigenvector, denoted by $\kett{\emptyset} = \ket{\phi}_1\ot \cdots \ot \ket{\phi}_n$, such that $\rho_{[n]}$ is weight-block-diagonal with respect to $\kett{\emptyset}$ and only supported on the weight-zero and weight-one subspaces.

    Then there exists an $n$-tuple of $d\times d$ PSD matrices $(X_1, \ldots, X_n)$ such that up to a local unitary change of basis (which fixes $\kett{\emptyset}$), we have
    \begin{equation}
        \rho_{[n]} \simeq \Gamma(X_1, \ldots, X_n).
    \end{equation}
\end{lem}
\begin{proof}
    Since $\rho_{[n]}$ is supported only on the weight-zero and weight-one subspaces and is block-diagonal with respect to these subspaces, the eigendecomposition of $\rho_{[n]}$ has the form:
    \begin{equation}
        \rho_{[n]} = \lambda_1 \kett{\emptyset}\braa{\emptyset} + \sum_{k=1}^{d} \lambda_{k+1} \ket{v_{k+1}}\bra{v_{k+1}}
    \end{equation}
    where the eigenvector $\ket{v_{k+1}}$ belongs to the weight-one subspace and thus has the form
    \begin{equation}
        \ket{v_{k+1}} = \sum_{p=1}^{d} \sum_{i=1}^{d} V_{p,i,k} \kett{i_p}
    \end{equation}
    for some unknown $n \times d \times d$ tensor of coefficients denoted by $V_{p,i,k}$ and where $\kett{i_p}$ is given by \cref{eq:weight_one} taken with respect to some $n$-tuple of bases where $\ket{0}_p \coloneqq \ket{\phi}_p$.
    Now, for each party $p$, let $M_p$ be the $d \times d$ matrix defined by
    \begin{equation}
        (M_p)_{i,k} \coloneqq \lambda_{k+1}^{\frac{1}{2}} V_{p,i,k}.
    \end{equation}
    With this definition, one obtains
    \begin{equation}
        \braa{i_p} \rho_{[n]} \kett{j_q} = \sum_{k=1}^{d} \lambda_{k+1} V_{p,i,k} V^{*}_{q,j,k} = \sum_{k=1}^{d} (M_p)_{i,k}(M_q)^{*}_{j,k} = \bra{i}_pM_{p}M_{q}^{\dagger} \ket{j}_q
    \end{equation}
    Now by a polar decomposition, for each party $p$, there exists a PSD $d\times d$ matrix $X_p$ and an $d \times d$ unitary $U_p$ such that
    \begin{equation}
        M_p = U_pX_p^{\frac{1}{2}},
    \end{equation}
    and therefore,
    \begin{equation}
        \braa{i_p} \rho_{[n]} \kett{j_q} = \bra{i}_pU_{p} X_{p}^{\frac{1}{2}} X_{q}^{\frac{1}{2}} U_{q}^{\dagger} \ket{j}_q.
    \end{equation}
    Letting $\ket{i}_p' = U_{p}^{\dagger} \ket{i}_p$ be a modified basis for party $p$ and letting
    \begin{equation}
        \s U_{[n]} \coloneqq (1 \oplus U_1) \ot (1 \oplus U_2) \ot \cdots \ot (1 \oplus U_n),
    \end{equation}
    we observe
    \begin{equation}
        \braa{i_p}' \s U_{[n]}^{\dagger} \rho_{[n]} \s U_{[n]} \kett{j_q}' = \braa{i_p} \rho_{[n]} \kett{j_q} = \bra{i}_pU_{p} X_{p}^{\frac{1}{2}} X_{q}^{\frac{1}{2}} U_{q}^{\dagger} \ket{j}_q = \braa{i_p}' X_{p}^{\frac{1}{2}} X_{q}^{\frac{1}{2}} \ket{j_{q}},
    \end{equation}
    which implies $\s U_{[n]}^{\dagger} \rho_{[n]} \s U_{[n]} = \Gamma(X_1, \ldots, X_p)$ as claimed.
\end{proof}

\section{Asymptotics of \texorpdfstring{$\boldsymbol{SU(d)}$ $\boldsymbol{6j}$}{SU(d) 6j} operators}

{\color{blue}
The purpose of this section is to derive some upper and lower bounds on generalized $6j$ symbol for $\SU(d)$ in relation to the existence or non-existence of Hermitian operators satisfying the tetrahedral relations \cref{eq:tetrahedral_linear_constraints}.
This section essentially constitutes a generalization of the proof techniques in the ``quantum-information-theoretic'' approach taken in \cite{christandl2008quantum}.

\subsection{Schur-Weyl decomposition \& spectrum estimation}

Throughout this section, we make use of the following decomposition of a tensor-power of a $d$-dimensional complex vector space.
\begin{prop}[Schur-Weyl decomposition]
    Let $n$ and $d$ be positive integers.
    Then the decomposition of $(\mathbb C^{d})^{\ot n}$ into irreducible representations of $S_n \times \SU(d)$, known as the \textit{Schur-Weyl} decomposition is given by
    \begin{equation}
        \label{eq:schur_weyl_decomp}
        (\mathbb C^{d})^{\ot n} \cong \bigoplus_{\lambda \vdash_{d} n} W_{\lambda} \ot V_{\lambda}^{d}
    \end{equation}
    where the direcy sum is indexed by Young diagrams $\lambda$ with $n$ boxes and at most $d-1$ rows.
\end{prop}
The following result constrains the associated Schur-Weyl decomposition of the $n$-th tensor power, $X^{\ot n}$, of a positive semidefinite Hermitian matrix $X$ \cite{christandl2008quantum,keyl2001estimating}.
\begin{prop}[Spectrum estimation]
    \label{prop:spec_est}
    Let $P_{\lambda}$ denote the Hermitian projection operator associated to the isotypic subspace indexed by $\lambda \vdash_{d} n$ in \cref{eq:schur_weyl_decomp} and let $X$ be a positive semidefinite matrix with eigenvalues $\chi$.
    Then
    \begin{equation}
        \Tr(P_{\lambda} X^{\ot n}) \leq \abs{\chi}^{n} \dim(V_{\lambda}^{d}) \exp(-n \kl{\bar \lambda}{\bar \chi}),
    \end{equation}
    where $\kl{\bar \lambda}{\bar \chi}$ denotes the Kullbeck-Leibler divergence of $\bar \lambda$ from $\bar \chi$.
\end{prop}
\begin{proof}
    See \cite{keyl2001estimating} for a proof (noting that the $\abs{\chi}^{n}$ term simply accounts for the possibility that $\Tr(X) \neq 1$ because $\Tr(X) = \abs{\chi}$).
\end{proof}

\begin{defn}
    \label{defn:Q_quantity}
    Let $(A,B,C,D,E,F)$ be a tetrahedral sextuple of non-zero positive semidefinite $d\times d$ Hermitian matrices with eigenvalues $(\alpha,\beta,\gamma,\delta,\epsilon,\phi)$
    Let $x,y,z$ be arbitrary positive integers and define
    \begin{equation}
        \Omega \coloneqq \frac{A^{\ot x}\ot B^{\ot y} \ot D^{\ot z}}{\abs{\alpha}^{x} \abs{\beta}^{y} \abs{\delta}^{z}}.
    \end{equation}
    Let $(\alpha', \beta', \gamma', \delta', \epsilon', \phi')$ be Young diagrams with sizes determined by \cref{eq:xyz_sized_young_diagrams}, and let $\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'}$ and $\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'}$ be defined accordining to \cref{eq:commuting_extended_projs}.
    Define $Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'} \in \mathbb C$ by
    \begin{equation}
        Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}
        \coloneqq 
        \Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'} \Omega \Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'}).
    \end{equation}

\end{defn}
\begin{prop}
    Let everything be as in \cref{defn:Q_quantity}.
    Then
    \begin{equation}
        \label{eq:Q_alt}
        Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'} 
        = 
        \Tr(\Pi_{\alpha'}\Pi_{\beta'}\Pi_{\gamma'}\Pi_{\delta'} \Pi_{\epsilon'}\Pi_{\phi'}\Omega),
    \end{equation}
    and therefore
    \begin{equation}
        \label{eq:completeness_relation}
        1 = \Tr(\Omega) = 
        \sum_{\alpha'\vdash_d x} 
        \sum_{\beta'\vdash_d y} 
        \sum_{\gamma'\vdash_d x+y} 
        \sum_{\delta'\vdash_d z} 
        \sum_{\epsilon'\vdash_d y+z} 
        \sum_{\phi'\vdash_d x+y+z} 
        Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}
    \end{equation}
    Moreover, the magnitude of $Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}$ can be upper bounded by either
    \begin{equation}
        \label{eq:Q_pre_upper_bound}
        \abs{Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}}
        \leq \left(\Tr(\Pi_{\alpha'}\Omega)\Tr(\Pi_{\beta'} \Omega)\Tr(\Pi_{\gamma'}\Omega)\Tr(\Pi_{\delta'}\Omega)\Tr(\Pi_{\epsilon'}\Omega)\Tr(\Pi_{\phi'}\Omega)\right)^{\frac{1}{6}}.
    \end{equation}
    or
    \begin{equation}
        \label{eq:Q_sixj_upper_bound}
        \abs{Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}} \leq \norm{\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'}\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'}}_{\infty}.
    \end{equation}
\end{prop}
\begin{proof}
    The proof of \cref{eq:Q_alt} and \cref{eq:completeness_relation} follows from elementary properties of the projection operators defined in \cref{lem:schur_weyl_presentation}.
    Moreover, \cref{eq:Q_sixj_upper_bound} follows because $\Omega$ is positive semidefinite and trace one.
    To prove \cref{eq:Q_pre_upper_bound}, consider an application of the Cauchy-Schwarz inequality to the Hilbert-Schmidt inner product, $\braket{X,Y} = \Tr(X^{\dagger} Y)$, whenever $X = \Omega^{\frac{1}{2}}\Pi_1$ and $Y = \Omega^{\frac{1}{2}}\Pi_2$ (where $\Pi_1$ and $\Pi_2$ are generic Hermitian projectors and $\Omega$ is positive semidefinite):
    \begin{align}
        \label{eq:biproj_ineq}
        \begin{split}
            &\abs{\Tr(\Pi_1 \Omega \Pi_2)}^2 
            = \abs{\Tr((\Omega^{\frac{1}{2}}\Pi_1)^{\dagger}(\Omega^{\frac{1}{2}}\Pi_2))}^2
            = \abs{\braket{\Omega^{\frac{1}{2}}\Pi_1, \Omega^{\frac{1}{2}}\Pi_2}}^{2}\\
            &\qquad\leq \braket{\Omega^{\frac{1}{2}}\Pi_1, \Omega^{\frac{1}{2}}\Pi_1}\braket{\Omega^{\frac{1}{2}}\Pi_2, \Omega^{\frac{1}{2}}\Pi_2}
            = \Tr(\Pi_1 \Omega)\Tr(\Pi_2 \Omega)
        \end{split}
    \end{align}
    Letting $\Pi_1 = \Pi_{\phi'}\Pi_{\alpha'}\Pi_{\beta'}$ and $\Pi_2 = \Pi_{\gamma'} \Pi_{\delta'} \Pi_{\epsilon'}$ and applying \cref{eq:biproj_ineq} yields 
    \begin{align}
        \begin{split}
            \abs{Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}}^{6} 
            &=\abs{\Tr[(\Pi_{\phi'}\Pi_{\alpha'}\Pi_{\beta'}) \Omega (\Pi_{\gamma'} \Pi_{\delta'} \Pi_{\epsilon'})]}^6\\
            &\leq 
            \Tr(\Pi_{\phi'}\Pi_{\alpha'}\Pi_{\beta'} \Omega)^3
            \Tr(\Pi_{\gamma'}\Pi_{\delta'}\Pi_{\epsilon'} \Omega)^3\\
            &\leq 
            \Tr(\Pi_{\phi'}\Omega)\Tr(\Pi_{\alpha'}\Omega)\Tr(\Pi_{\beta'} \Omega)
            \Tr(\Pi_{\gamma'}\Omega)\Tr(\Pi_{\delta'}\Omega)\Tr(\Pi_{\epsilon'} \Omega).
        \end{split}
    \end{align}
\end{proof}

\begin{thm}
    \label{thm:Q_upper_bound}
    Let everything be as in \cref{defn:Q_quantity} except fix $x,y,z$ to be
    \begin{equation}
        x \coloneqq \ceil{n \abs{\alpha}},
        \qquad
        y \coloneqq \ceil{n \abs{\beta}},
        \qquad
        z \coloneqq \ceil{n \abs{\delta}},
    \end{equation}
    where $n$ is a sufficiently large positive integer ensuring that $x,y$ and $z$ are non-zero.
    Then
    \begin{equation}
        \abs{Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}} \leq p(n) \exp(-n I^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}_{\alpha\beta\gamma\delta\epsilon\phi}/6).
        \label{eq:Q_upper_bound}
    \end{equation}
    where $p(n)$ is a polynomial in $n$ and where
    \begin{equation}
        I^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}_{\alpha\beta\gamma\delta\epsilon\phi}
        =\abs{\alpha} \kl{\bar \alpha'}{\bar \alpha}
        +\abs{\beta} \kl{\bar \beta'}{\bar \beta}
        +\abs{\gamma} \kl{\bar \gamma'}{\bar \gamma}
        +\abs{\delta} \kl{\bar \delta'}{\bar \delta}
        +\abs{\epsilon} \kl{\bar \epsilon'}{\bar \epsilon}
        +\abs{\phi} \kl{\bar \phi'}{\bar \phi}.
    \end{equation}
    \textcolor{red}{TODO: Is this relative entropy deviation measure conceptually useful or is it best to relax the deviation measure to its geometric analogue using Pinsker's inequality?}
\end{thm}
\begin{proof}
    To begin the proof, we consider $x,y$ and $z$ to be arbitrary positive integers.
    For each $\lambda' \in \{\alpha', \beta', \gamma', \delta', \epsilon', \phi'\}$, it is possible to upper-bound $\Tr(\Pi_{\lambda'} \Omega)$ using the spectrum estimation technique \cref{prop:spec_est}.
    For a detailed example, we will consider the bound on $\Tr(\Pi_{\gamma'} \Omega)$.
    \begin{align}
        \Tr(\Pi_{\gamma'} \Omega)
        = \frac{\Tr[(P_{\gamma'} \ot \id^{\ot z})(A^{\ot x}\ot B^{\ot y} \ot D^{\ot z})]}{\abs{\alpha}^{x}\abs{\beta}^{y}\abs{\delta}^{z}}
        = \frac{\Tr[P_{\gamma'}(A^{\ot x}\ot B^{\ot y})]\abs{\delta}^{z}}{\abs{\alpha}^{x}\abs{\beta}^{y}\abs{\delta}^{z}}.
    \end{align}
    Using the assumption that $C = A+B$, we have $C^{\ot x+y} = (A+B)^{\ot x+y}$ which implies
    \begin{equation}
        \Tr(P_{\gamma'} C^{\ot x+y}) = \sum_{k=0}^{x+y} \binom{x+y}{k} \Tr(P_{\gamma'} (A^{\ot k} \ot B^{\ot x+y-k}).
    \end{equation}
    Here we have used the $S_{x+y}$-equivariance of $P_{\gamma'}$.
    Since each of the summands above are non-negative, we conclude for all $0 \leq k \leq x+y$ that
    \begin{equation}
        \Tr(P_{\gamma'} C^{\ot x+y}) \geq \binom{x+y}{k} \Tr(P_{\gamma'} (A^{\ot k} \ot B^{\ot x+y-k}).
    \end{equation}
    Letting $k = x$ and rearranging yields
    \begin{equation}
        \Tr(P_{\gamma'} (A^{\ot x} \ot B^{\ot y})) \leq \frac{x!y!}{(x+y)!}\Tr(P_{\gamma'} C^{\ot x+y}).
    \end{equation}
    Furthermore, spectrum estimation (\cref{prop:spec_est}) yields
    \begin{align}
        \Tr(\Pi_{\gamma'} \Omega)
        &\leq \frac{1}{\abs{\alpha}^{x}\abs{\beta}^{y}}\frac{x!y!}{(x+y)!}\Tr[P_{\gamma'}C^{\ot x+y}], \qquad \\
        &\leq \frac{(\abs{\alpha}+\abs{\beta})^{x+y}}{\abs{\alpha}^{x}\abs{\beta}^{y}}\frac{x!y!}{(x+y)!}\dim(V_{\gamma'}^{d})\exp(-(x+y)\kl{\bar \gamma'}{\bar \gamma}).
    \end{align}
    Analogous calculations for $\lambda' \in \{\alpha', \beta', \gamma', \delta', \epsilon', \phi'\}$ produce the following set of upper bounds:
    \begin{align}
        \Tr(\Pi_{\alpha'} \Omega)
        &\leq \dim(V_{\alpha'}^{d}) \exp(- x \kl{\bar \alpha'}{\bar \alpha}), \\
        \Tr(\Pi_{\beta'} \Omega)
        &\leq \dim(V_{\beta'}^{d}) \exp(- y \kl{\bar \beta'}{\bar \beta}), \\
        \Tr(\Pi_{\gamma'} \Omega)
        &\leq \frac{(\abs{\alpha}+\abs{\beta})^{x+y}}{\abs{\alpha}^{x}\abs{\beta}^{y}}\frac{x!y!}{(x+y)!}\dim(V_{\gamma'}^{d}) \exp(- (x+y) \kl{\bar \gamma'}{\bar \gamma}), \\
        \Tr(\Pi_{\delta'} \Omega)
        &\leq \dim(V_{\delta'}^{d}) \exp(- z \kl{\bar \delta'}{\bar \delta}), \\
        \Tr(\Pi_{\epsilon'} \Omega)
        &\leq \frac{(\abs{\alpha}+\abs{\beta}+\abs{\delta})^{x+y+z}}{\abs{\alpha}^{x}\abs{\beta}^{y}\abs{\delta}^{z}}\frac{x!y!z!}{(x+y+z)!}\dim(V_{\epsilon'}^{d}) \exp(- (x+y+z) \kl{\bar \epsilon'}{\bar \epsilon}), \\
        \Tr(\Pi_{\phi'} \Omega)
        &\leq \frac{(\abs{\beta}+\abs{\delta})^{x+y+z}}{\abs{\beta}^{y}\abs{\delta}^{z}}\frac{y!z!}{(y+z)!} \dim(V_{\phi'}^{d}) \exp(- (y+z) \kl{\bar \phi'}{\bar \phi}).
    \end{align}
    Our objective now will be to use the given values for $x,y,z$ to prove the multinomial prefactors appearing in the bounds for $\gamma'$, $\epsilon'$, and $\phi'$ reasonably small compared to $n$. 
    An application of \cref{lem:multinomial_dist_lower_bound} (when $k=2$) implies the following bound
    \begin{equation}
        \frac{(\abs{\alpha}+\abs{\beta})^{x+y}}{\abs{\alpha}^{x}\abs{\beta}^{y}}\frac{x!y!}{(x+y)!}
        \leq e^{4}\left(n (\abs{\alpha} + \abs{\beta})+3\right)^2.
    \end{equation}
    Moreover, using \cite[Equation 7]{christandl2006spectra}, we have
    \begin{equation}
        \dim(V_{\gamma'}^{d}) \leq (\abs{\gamma'}+1)^{d(d-1)/2} = (x+y+1)^{d(d-1)/2} < (n (\abs{\alpha}+\abs{\beta}) + 3)^{d(d-1)/2}.
    \end{equation}
    Altogether, we can relax the bound on $\Tr(\Pi_{\gamma'} \Omega)$ to obtain
    \begin{equation}
        \Tr(\Pi_{\gamma'} \Omega) \leq
        e^{4}(n\abs{\epsilon}+3)^{\frac{d(d-1)}{2}+2}\exp\left(-n\abs{\gamma}\kl{\bar \gamma'}{\bar \gamma}\right).
    \end{equation}
    Similar calculations for $\Tr(\Pi_{\epsilon'} \Omega)$ yields (noting that $k=3$ when applying \cref{lem:multinomial_dist_lower_bound}):
    \begin{equation}
        \Tr(\Pi_{\epsilon'} \Omega) \leq e^{16}(n\abs{\epsilon}+4)^{\frac{d(d-1)}{2}+3}\exp\left(-n\abs{\epsilon}\kl{\bar \epsilon'}{\bar \epsilon}\right).
    \end{equation}
    It is also not too difficult to see that the polynomial in $n$ term associated to the upper bound on $\Tr(\Pi_{\epsilon'} \Omega)$ above is the largest such subexponential term and thus \cref{eq:Q_upper_bound} indeed holds if we take $p(n)$ to be
    \begin{equation}
        p(n) = e^{16}(n\abs{\epsilon}+4)^{\frac{d(d-1)}{2}+3}.
    \end{equation}
\end{proof}

\begin{cor}
    \label{cor:seq_ud_recoupling_norm_lower_bound}
    Let $(\alpha,\beta,\gamma,\delta,\epsilon,\phi)$ be the eigenvalues of non-zero positive semidefinite tetrahedral Hermitian matrices $(A,B,C,D,E,F)$.
    Then there exists a sextuple of sequences of Young diagrams $(\alpha'_n, \beta'_n, \gamma'_n, \delta'_n, \epsilon'_n, \phi'_n)$ of sizes 
    \begin{equation}
        \label{eq:ceiling_sized_young_diagrams}
        \begin{alignedat}{3}
            &\alpha'_n \vdash_{d} \ceil{n\abs{\alpha}}, \qquad 
            &&\beta'_n \vdash_{d} \ceil{n\abs{\beta}}, \qquad
            &&\gamma'_n \vdash_{d} \ceil{n\abs{\alpha}}+\ceil{n\abs{\beta}}, \qquad\\
             &\delta'_n \vdash_{d} \ceil{n\abs{\delta}}, \qquad
            &&\epsilon'_n \vdash_{d} \ceil{n\abs{\alpha}}+\ceil{n\abs{\beta}}+\ceil{n\abs{\delta}}, \qquad
            &&\phi'_n \vdash_{d} \ceil{n\abs{\beta}}+\ceil{n\abs{\delta}},
        \end{alignedat}
    \end{equation}
    satisfying (i) the convergence conditions
    \begin{equation}
        \label{eq:sextuple_convergence}
        \begin{alignedat}{3}
            &\lim_{n\to\infty} \bar \alpha'_n = \bar \alpha, \qquad
            &&\lim_{n\to\infty} \bar \beta'_n = \bar \beta, \qquad
            &&\lim_{n\to\infty} \bar \gamma'_n = \bar \gamma, \\
            &\lim_{n\to\infty} \bar \delta'_n = \bar \delta, \qquad
            &&\lim_{n\to\infty} \bar \epsilon'_n = \bar \epsilon, \qquad
            &&\lim_{n\to\infty} \bar \phi'_n = \bar \phi,
        \end{alignedat}
    \end{equation}
    and (ii) the lower bound 
    \begin{equation}
        \label{eq:seq_ud_sixj_lower_bound}
        \norm{\tightsixj{ \alpha'_n & \beta'_n & \gamma'_n \\ \delta'_n &  \epsilon'_n &  \phi'_n}}_{\infty}
        \geq \frac{1}{(n \abs{\epsilon}+4)^{6(d-1)}}.
    \end{equation}
\end{cor}
\begin{proof}
    For each $n$, it suffices to pick $(\alpha'_n, \beta'_n, \gamma'_n, \delta'_n, \epsilon'_n, \phi'_n)$ be the sextuple which maximizes the value of $\abs{Q_{A,B,C,D,E,F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}}$.
    Such a choice will necessarily satisfy a lower bound due to the completeness relation \cref{eq:completeness_relation}.
    Specifically,
    \begin{equation}
        \label{eq:Q_lower_bound}
        \abs{Q_{A,B,C,D,E,F}^{\alpha'_n\beta'_n\gamma'_n\delta'_n\epsilon'_n\phi'_n}} \geq \frac{1}{q(n)}
    \end{equation}
    where $q(n)$ is any upper bound on the number of possible summands in \cref{eq:completeness_relation} viewed as a function of $n$.
    As there are at most $(k+1)^{d-1}$ Young diagrams with $d-1$ rows and size at most $k$, letting
    \begin{equation}
        q(n) = (n \abs{\epsilon}+4)^{6(d-1)}
    \end{equation}
    ensures \cref{eq:Q_lower_bound} holds because the size of $\epsilon'_n$ is at most $k=n\abs{\epsilon} + 3$.
    Moreover, applying \cref{eq:Q_upper_bound} yields
    \begin{equation}
        I^{\alpha'_n\beta'_n\gamma'_n\delta'_n\epsilon'_n\phi'_n}_{\alpha\beta\gamma\delta\epsilon\phi}
        \leq \frac{6}{n}\left(\log q(n) + \log p(n)\right) \stackrel{n\to\infty}{\longrightarrow} 0
    \end{equation}
    and thus \cref{eq:sextuple_convergence} holds because $\kl{\bar \lambda'}{\bar \lambda} \geq 0$ with equality if and only if $\bar \lambda' = \bar \lambda$.
    Then \cref{eq:seq_ud_sixj_lower_bound} follows from \cref{eq:Q_lower_bound} combined with \cref{eq:Q_sixj_upper_bound} and \cref{eq:ud_sixj_prod_proj_operator_norm}.
\end{proof}

\begin{thm}
    Let $(\alpha, \beta, \gamma, \delta, \epsilon, \phi)$ be non-negative non-zero eigenvalues which are not tetrahedral. 
    Then for all sequences of Young diagrams $(\alpha'_n, \beta'_n, \gamma'_n, \delta'_n, \epsilon'_n, \phi'_n)$ satisfying \cref{eq:ceiling_sized_young_diagrams} and \cref{eq:sextuple_convergence}, we have
    \begin{equation}
        \label{eq:seq_ud_sixj_upper_bound}
        \norm{\tightsixj{ \alpha'_n & \beta'_n & \gamma'_n \\ \delta'_n &  \epsilon'_n &  \phi'_n}}_{\mathrm{HS}}^{2}
        \leq \exp(-n I_{*}^{\alpha\beta\gamma\delta\epsilon\phi} /6 + O(\log n)).
    \end{equation}
    where $I_{*}^{\alpha\beta\gamma\delta\epsilon\phi} > 0$ quantifies the deviation of $(\alpha, \beta, \gamma, \delta, \epsilon, \phi)$ from being tetrahedral:
    \begin{equation}
        I_{*}^{\alpha\beta\gamma\delta\epsilon\phi} = \min_{\substack{\tilde \alpha \tilde \beta \tilde \gamma\tilde\delta \tilde \epsilon \tilde \phi\\\mathrm{tetrahedral}}}I_{\tilde \alpha\tilde \beta\tilde \gamma\tilde \delta\tilde \epsilon\tilde \phi}^{\alpha\beta\gamma\delta\epsilon\phi}.
    \end{equation}
\end{thm}
\begin{proof}
    Let $\diff X$ be the normalized Hilbert-Schmidt measure over $d\times d$ density matrices and let $k$ be a positive integer.
    Then define an operator $\Gamma_{k}$ on $(\mathbb C^{d})^{\ot k}$ by
    \begin{equation}
        \Gamma_{k} \coloneqq \int \diff X X^{\ot k}.
    \end{equation}
    As $\Gamma_{k}$ is known to be both $S_k$ and $\SU(d)$ equivariant, by Schur's lemma, $\Gamma_{k}$ can be expressed as 
    \begin{equation}
        \Gamma_{k} = \sum_{\lambda' \vdash_d k} r_{\lambda'} P_{\lambda'}
    \end{equation}
    with coefficients $r_{\lambda'}$ that can be computed \cite{zyczkowski2001induced,walter2014multipartite} to obtain
    \begin{equation}
        r_{\lambda'} = \frac{\dim(V_{\lambda'}^{d})}{\dim(W_{\lambda'})\dim(V_{(k)}^{d^2})}.
    \end{equation}
    Now let $(\alpha', \beta', \gamma', \delta', \epsilon', \phi')$ be arbitary Young diagrams with respective sizes $(x,y,x+y,z,x+y+z,y+z)$ where $x,y,z$ are determined by a positive integer $n$ as in \cref{thm:Q_upper_bound}.
    Using \cref{eq:ud_sixj_prod_proj_hs_norm}, we obtain the formula
    \begin{align}
        \norm{\tightsixj{ \alpha' & \beta' & \gamma' \\ \delta' &  \epsilon' &  \phi'}}_{\mathrm{HS}}^2 
        &=\frac{\Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'}\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'})}{\dim(V_{\epsilon'}^{d})\dim(W_{\alpha'})\dim(W_{\beta'})\dim(W_{\delta'})} \\
        &=R_{\alpha',\beta',\delta';\epsilon'}\Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'} (\Gamma_{x}\ot \Gamma_{y} \ot \Gamma_{z})\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'})
    \end{align}
    where the cooefficient $R_{\alpha',\beta',\delta';\epsilon'}$ is a ratio of dimensions:
    \begin{equation}
        R_{\alpha',\beta',\delta';\epsilon'} \coloneqq \frac{\dim(V_{(x)}^{d^2})\dim(V_{(y)}^{d^2})\dim(V_{(z)}^{d^2})}{\dim(V_{\epsilon'}^{d})\dim(V^{d}_{\alpha'})\dim(V^{d}_{\beta'})\dim(V^{d}_{\delta'})}.
    \end{equation}
    Moreover, by the definition of $\Gamma_k$, we have
    \begin{align}
        &\Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'} (\Gamma_{x}\ot \Gamma_{y} \ot \Gamma_{z})\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'})\\
        &\qquad=
        \int \diff X \int \diff Y \int \diff Z \Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'} (X^{\ot x} \ot Y^{\ot y} \ot Z^{\ot z})\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'}),
    \end{align}
    where the integral is taken over a triple of $d\times d$ \textit{density matrices} $X,Y,Z$.
    For each integrand, define a tetrahedral sextuple $(\tilde A,\tilde B,\tilde C,\tilde D,\tilde E,\tilde F)$ as follows
    \begin{equation}
        \begin{alignedat}{3}
            &\tilde A \coloneqq \abs{\alpha}X, \qquad 
            &&\tilde B \coloneqq \abs{\beta}Y, \qquad 
            &&\tilde C \coloneqq \abs{\alpha}X+\abs{\beta}Y, \qquad \\
            &\tilde D \coloneqq \abs{\delta}Z, \qquad 
            &&\tilde E \coloneqq \abs{\alpha}X+\abs{\beta}Y+\abs{\gamma}Z, \qquad 
            &&\tilde F \coloneqq \abs{\beta}Y + \abs{\delta}Z.
        \end{alignedat}
    \end{equation} 
    Furthermore, let $(\tilde \alpha, \tilde \beta, \tilde \gamma, \tilde \delta, \tilde \epsilon, \tilde \phi)$ be the eigenvalues of $(\tilde A, \tilde B, \tilde C, \tilde D, \tilde E, \tilde F)$.
    In this notation, the integrand above is equal to
    \begin{equation}
        Q_{\tilde A, \tilde B, \tilde C, \tilde D, \tilde E, \tilde F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi} = \Tr(\Pi^{\phi'}_{\alpha'\beta'\delta'\epsilon'} (X^{\ot x} \ot Y^{\ot y} \ot Z^{\ot z})\Pi^{\gamma'}_{\alpha'\beta'\delta'\epsilon'})
    \end{equation}
    Moreover, by \cref{eq:Q_sixj_upper_bound} we have
    \begin{equation}
        \abs{Q_{\tilde A,\tilde B,\tilde C,\tilde D,\tilde E,\tilde F}^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}} \leq p(n) \exp(-n I^{\alpha'\beta'\gamma'\delta'\epsilon'\phi'}_{\tilde \alpha \tilde \beta\tilde \gamma\tilde \delta\tilde \epsilon\tilde \phi}/6).
    \end{equation}
    Under the assumption that $(\alpha, \beta, \gamma, \delta, \epsilon, \phi)$ are not tetrahedral, we conclude that
    \begin{equation}
        (\alpha, \beta, \gamma, \delta, \epsilon, \phi) \neq (\tilde \alpha, \tilde \beta, \tilde \gamma, \tilde \delta, \tilde \epsilon, \tilde \phi)
    \end{equation}
    and therefore $I_{\tilde \alpha\tilde \beta\tilde \gamma\tilde \delta\tilde \epsilon\tilde \phi}^{\alpha\beta\gamma\delta\epsilon\phi} > 0$.
    \textcolor{red}{TODO: Figure out how to describe the convergence, is there a triangle inequality for relative entropies? Maybe we need to relax the rate function $I$ to its geometric analogue using Pinsker's inequality, then everything works.}
\end{proof}

\subsection{The asymptotics of irrep dimensions}

It will be useful for what follows to better understand how the dimension, $\dim(V_{\lambda}^{d})$, of the irreducible representation $V_{\lambda}^{d}$ of $\SL(d)$ grows as $d$ becomes much larger $\abs{\lambda}$ while $\abs{\lambda}$ itself is much larger than $\ell(\lambda)$.
\Citeauthor{itzykson1966unitary} provide the following estimate for $\dim(V_{\lambda}^{d})$ \cite{itzykson1966unitary}.
\begin{lem}
    \label{lem:in_estimate}
    Let $\lambda$ be an positive integer partition of length $\ell \coloneqq \ell(\lambda)$, let $d \geq \ell$ be a positive integer and let $\rho^d \coloneqq (d-1, d-2, \ldots, 2, 1)$ be the Weyl-vector for $\SL(d)$.
    The following approximation holds when $d \gg \abs{\lambda}^2$ and $\lambda_{\ell} \gg \ell$:
    \begin{equation}
        \label{eq:in_estimate}
        \dim(V_{\lambda}^{d}) = \dim(V_{\lambda}^{\ell}) \times \prod_{i=1}^{\ell}\frac{\rho^{\ell}_{i}!}{(\lambda_i + \rho^{\ell}_{i}) !}  \times d^{\abs{\lambda}} \times \left(1 + O\left(\frac{\abs{\lambda}^{2}}{d}\right)\right).
    \end{equation}
\end{lem}
\begin{proof}
    Weyl's character formula can be used to derive the following formula for $\dim(V_{\lambda}^{d})$ in terms of the Young diagram $\lambda$ and the dimension $d$ \cite{itzykson1966unitary}:
    \begin{equation}
        \dim(V_{\lambda}^{d}) = \frac{\Delta(\lambda + \rho^d)}{\Delta(\rho^d)} = \prod_{1 \leq i<j\leq d} \frac{(\lambda_i+\rho^d_i)-(\lambda_j+\rho^d_j)}{\rho^d_i - \rho^d_j} = \prod_{1 \leq i<j\leq d} \frac{(\lambda_i-i)-(\lambda_j-j)}{j-i},
    \end{equation}
    where $\Delta(x) = \prod_{i < j}(x_i-x_j)$ is the Vandermonde determinant.
    Isolating the terms above which depend on $d$ from those that do not yields
    \begin{equation}
        \label{eq:dim_separation_large_d}
        \dim(V_{\lambda}^{d}) = \dim(V_{\lambda}^{\ell}) \times \prod_{i=1}^{\ell}\frac{\rho^{\ell}_{i}!}{(\lambda_i + \rho^{\ell}_{i}) !} \times \prod_{i=1}^{\ell} \frac{(\lambda_i+\rho^{d}_i)!}{\rho^{d}_i!}.
    \end{equation}
    Given positive integers $a$ and $b$ such that $b \gg a^2$, one obtains the following approximation:
    \begin{align}
        \begin{split}
            &\log\left(\frac{(a+b)!}{b!}\right) 
            = \log \prod_{j=1}^{a} (b+j)
            = \sum_{j=1}^{a} \log (b+j) \\
            &\quad= a \log b + \sum_{j=1}^{a} \log \left(1+\frac{j}{b}\right)
            = a \log b + O\left(\frac{a^2}{b}\right).
        \end{split}
    \end{align}
    The assumption that $d \gg \abs{\lambda}^2$ and $i \leq \ell \ll \lambda_{\ell} \leq \abs{\lambda}$ also implies $d - i = \rho^{d}_i \gg \lambda_i^2$.
    Therefore,
    \begin{align}
        \log\left(\frac{(\lambda_i+\rho^{d}_i)!}{\rho^{d}_i!}\right) 
        &= \lambda_i\log\rho^{d}_i + O\left(\frac{\lambda_i^2}{\rho^{d}_i}\right).
    \end{align}
    The assumptions also imply the following simplification
    \begin{align}
        \log\left(\frac{(\lambda_i+\rho^{d}_i)!}{\rho^{d}_i!}\right) 
        &= \lambda_i\log d + O\left(\frac{\abs{\lambda}^2}{d}\right)
    \end{align}
    which, when applied to \cref{eq:dim_separation_large_d}, proves the claim.
\end{proof}
Using \cref{lem:in_estimate}, in can be seen that for fixed $\lambda$, $\dim(V_{\lambda}^{d})$ generally grows polynomially in $d$ with leading exponent $\abs{\lambda}$ and leading coefficient depending only on $\lambda$.
The objective of the following lemma is to characterize the asymptotics of this leading coefficient for large $\abs{\lambda}$ and even large $d$.
\textcolor{red}{TODO: Figure out a elegant way to present these ``double asymptotics'', maybe just epsilonifying everything, like $\max\{\ell/\abs{\lambda}, \abs{\lambda}/d\} \leq \epsilon$.}
\begin{lem}
    \label{lem:double_asymptotics}
    Let everything be as in \cref{lem:in_estimate}. Then
    \begin{equation}
        \log \dim (V_{\lambda}^{d}) = \abs{\lambda} + \abs{\lambda}\log(d) + s(\lambda) + O(\ell^2 \log \abs{\lambda}) + O\left(\frac{\abs{\lambda}^2}{d}\right)
    \end{equation}
    where $s(\lambda)$ is defined by
    \begin{equation}
        \label{eq:unnorm_shannon}
        s(\lambda) \coloneqq -\sum_{i=1}^{\ell(\lambda)} \lambda_i \log \lambda_i.
    \end{equation}
\end{lem}
\begin{proof}
    The proof proceeds by considering the multiplicands in \cref{eq:in_estimate} separately.
    First, we upper bound the dimension $\dim(V_{\lambda}^{\ell})$; for indices $i<j\leq \ell$, one has $(\lambda_i-i)-(\lambda_j-j) \leq \abs{\lambda} + 1$ as well as $j-i \geq 1$ which implies
    \begin{equation}
        \label{eq:dim_lambda_ell_bound}
        \dim(V_{\lambda}^{\ell}) = \prod_{1 \leq i<j\leq \ell} \frac{(\lambda_i-i)-(\lambda_j-j)}{j-i} \leq (\abs{\lambda}+1)^{\ell(\ell-1)/2},
    \end{equation}
    and thus $\log \dim(V_{\lambda}^{\ell}) = O(\ell^2 \log \abs{\lambda})$.

    Second, we apply Stirling's approximation \cite{robbins1955remark} to the second term in \cref{eq:in_estimate} under the given assumption that $\lambda_{\ell} \gg \ell$ (which implies $\rho^{\ell}_{i} \ll \lambda_{i}$),
    \begin{align}
        \log\left(\prod_{i=1}^{\ell}\frac{\rho^{\ell}_{i}!}{(\lambda_i + \rho^{\ell}_{i}) !}\right) 
        &= \sum_{i=1}^{\ell}\log \rho^{\ell}_i! - \sum_{i=1}^{\ell} \log (\lambda_i + \rho^{\ell}_i)!, \\
        &= \abs{\lambda} + \sum_{i=1}^{\ell} \left(\rho^{\ell}_i\log \rho^{\ell}_i - (\lambda_i + \rho^{\ell}_i)\log (\lambda_i + \rho^{\ell}_i)+ O(\log \lambda_i )\right), \\
        &= \abs{\lambda} - \sum_{i=1}^{\ell} \lambda_i\log \lambda_i + O( \ell \log \abs{\lambda} ), \\
        \label{eq:unnorm_shannon_term}
    \end{align}
    Combining \cref{eq:dim_lambda_ell_bound} and \cref{eq:unnorm_shannon_term} with \cref{lem:in_estimate} proves the claim.
\end{proof}
\begin{rem}
    Note that the quantity $s(\lambda)$ defined in \cref{eq:unnorm_shannon} resembles the Shannon entropy,
    \begin{equation}
        S(p) = -\sum_{i=1}^{\ell} p_i \log p_i
    \end{equation}
    of a probability distribution $p = (p_1, \ldots, p_\ell)$.
    Indeed, $s(\lambda)$ is related to the Shannon entropy of the \textit{normalization} of $\lambda$ by the formula
    \begin{equation}
        \label{eq:little_s_relation}
        s(\lambda) = \abs{\lambda} S\left(\frac{\lambda}{\abs{\lambda}}\right) - \abs{\lambda} \log \abs{\lambda}.
    \end{equation}
\end{rem}

\end{thm}
\begin{proof}

\begin{lem}
    \label{lem:asymptotic_dimension_ratio}
    Let $\gamma,\phi,\beta,\epsilon$ be a quadruple of Young diagrams such that 
    \begin{equation}
        \label{eq:balanced_sizes}
        \abs{\gamma}+\abs{\phi} = \abs{\beta}+\abs{\epsilon}
    \end{equation}
    and that $\abs{\epsilon} = \max\{\abs{\gamma}, \abs{\phi}, \abs{\beta}, \abs{\epsilon}\}$.
    Let $\ell \coloneqq \max\{\ell(\gamma), \ell(\phi), \ell(\beta), \ell(\epsilon) \}$ and let $d$ be a positive integer such that $d \geq \ell$.
    Then
    \begin{equation}
        \log\left(\frac{\dim(V_{\gamma}^d) \dim(V_{\phi}^d)}{\dim(V_{\beta}^d) \dim(V_{\epsilon}^d)}\right)
        = I(\gamma,\phi;\beta,\epsilon) + O(\ell^2 \log \abs{\epsilon}) + O\left(\frac{\abs{\epsilon}^2}{d}\right)
        \label{eq:asymptotic_dimension_ratio}
    \end{equation}
    where $I(\gamma,\phi;\beta,\epsilon)$ is the function
    \begin{equation}
        I(\gamma,\phi;\beta,\epsilon) \coloneqq s(\gamma) + s(\phi) - s(\beta) - s(\epsilon).
    \end{equation}
    Moreover, $I(k\gamma,k\phi;k\beta,k\epsilon) = k I(\gamma,\phi;\beta,\epsilon)$ for any $k > 0$.
\end{lem}
\begin{proof}
    The proof follows from four applications of \cref{lem:double_asymptotics}.
    The assumption that the sizes of the Young diagrams are balanced in the sense of \cref{eq:balanced_sizes} implies that the $\log(d)$ terms appearing in \cref{lem:double_asymptotics} cancel each other and one obtains \cref{eq:asymptotic_dimension_ratio}.
    
    Four applications of \cref{eq:little_s_relation} yields
    \begin{align}
        \begin{split}
            I(\gamma,\phi;\beta,\epsilon)
            &= s(\gamma) + s(\phi) - s(\beta) - s(\epsilon) \\
            &= \left\{\abs{\gamma}S(\bar \gamma) + \abs{\phi} S(\bar \phi) - \abs{\beta} S(\bar \beta) - \abs{\epsilon} S(\bar \epsilon)\right\} + \cdots \\
            &\quad\cdots- \left\{\abs{\gamma}\log \abs{\gamma} + \abs{\phi}\log \abs{\phi} - \abs{\beta}\log \abs{\beta} - \abs{\epsilon}\log \abs{\epsilon}\right\}.
        \end{split}
    \end{align}
    Under scaling $\lambda \mapsto k \lambda$ for any $k>0$, observe that $\overline{k\lambda} = \bar \lambda$ and $\abs{k\lambda} = k \abs{\lambda}$.
    Therefore the above expression implies
    \begin{equation}
        \label{eq:pseudo_ssa_scaling}
        I(k\gamma,k\phi;k\beta,k\epsilon) = kI(\gamma,\phi;\beta,\epsilon) + (\abs{\gamma}+\abs{\phi}-\abs{\beta}-\abs{\epsilon}) k\log k.
    \end{equation}
    Since \cref{eq:balanced_sizes} is assumed to hold, the coefficient in front of the $k\log k$ term appearing in \cref{eq:pseudo_ssa_scaling} vanishes and therefore $I(k\gamma,k\phi;k\beta,k\epsilon) = k I(\gamma,\phi;\beta,\epsilon)$ as claimed.
\end{proof}

\begin{thm}
    Let $(\alpha,\beta,\gamma,\delta,\epsilon,\phi)$ be the eigenvalues of Hermitian matrices $(A,B,C,D,E,F)$ satisfying \cref{eq:tetrahedral_linear_constraints}.
    Then necessarily $\abs{\beta}+\abs{\epsilon} = \abs{\gamma}+\abs{\phi}$ and furthermore
    \begin{equation}
        I(\gamma, \phi; \beta, \epsilon) \geq 0.
    \end{equation} 
\end{thm}

\section{Entropic identities \& inequalities}

In this section, we recall a handful of identities and inequalities concerning multinomial coefficients and the multinomial distribution in the context of Sanov's theorem \cite[Chapter 2]{dembo2010large}.
Let $q = (q_1,\ldots, q_k)$ be a probability distribution such that for all $i \in \{1, \ldots, k\}$, $n q_i$ is an integer. 
Let $M_{n}(q)$ denote the following multinomial coefficient:
\begin{equation}
    \label{eq:multinomial_coefficient}
    M_{n}(q) \coloneqq \binom{n}{nq_1, \ldots, nq_k} = \frac{n!}{(nq_1)!\cdots (nq_k)!},
\end{equation}
which can be bounded (see \cite[Lemma 2.1.8]{dembo2010large}) in terms of $\ent{q}$ by ,
\begin{equation}
    \label{eq:multinomial_coefficient_bound}
    (n+1)^{-k} e^{n \ent{q}} \leq M_{n}(q) \leq e^{n \ent{q}},
\end{equation}
where $\ent{q} = - \sum_{i=1}^{k} q_i \log q_i$ is the Shannon entropy of $q$.
Additionally, we have for any probability distribution $p = (p_1, \ldots, p_k)$,
\begin{equation}
    \label{eq:multinomial_probability}
    p_1^{nq_1} \cdots p_k^{nq_k} = e^{-n \left(\ent{q} + \kl{q}{p}\right)}
\end{equation}
where $\kl{q}{p} = \sum_{i=1}^{k} q_i \log q_i/p_i$ denotes the Kullbeck-Leibler divergence or relative entropy of $q$ from $p$ with the convention that $\kl{q}{p} = \infty$ if $q_i > 0$ whenever $p_i=0$.
Note that $\kl{q}{p} = 0$ if and only if $p=q$.
Combining \cref{eq:multinomial_coefficient_bound} with \cref{eq:multinomial_probability} places bounds on the multinomial distribution
\begin{equation}
    (n+1)^{-k} e^{-n \kl{q}{p}} \leq \binom{n}{nq_1, \ldots, nq_k} p_1^{nq_1} \cdots p_k^{nq_k} \leq e^{-n \kl{q}{p}}
\end{equation}
If needed, it is possible to relax these bounds further using Pinsker's inequality for $\kl{q}{p}$ in conjunction with its reverse (which only holds for $\kl{q}{p} < \infty$) \cite{gotze2019higher}:
\begin{equation}
    2 d_{\mathrm{TV}}^2(p,q) \leq \kl{q}{p} \leq \frac{4}{p_{\min}}d_{\mathrm{TV}}^2(p,q) 
\end{equation}
where
\begin{equation}
    \label{eq:pinskers}
    d_{\mathrm{TV}}(p,q) = \max_{1\leq j\leq k}\abs{p_j - q_j} = \frac{1}{2} \sum_{j=1}^{k}\abs{p_j - q_j}
\end{equation}
is known as the \textit{total variation distance} between $p$ and $q$.
Now for any distributon $p = (p_1, \ldots, p_k)$ and any positive integer $n$, it is possible to find another distribution $p' = (p'_1, \ldots, p'_k)$ such that (i) $np'_i$ is an integer for all $i \in \{1,\ldots, k\}$, (ii) $\kl{p'}{p} < \infty$ is finite, and (iii) $p'$ approximates $p$ in the sense that for some constant $C \geq 0$,
\begin{equation}
    \label{eq:tv_approx}
    d_{\mathrm{TV}}(p,p') \leq \frac{C}{n}.
\end{equation}
Concretely, one can choose $p'$ such that
\begin{equation}
    \label{eq:nint_approx}
    np'_j \coloneqq \mathrm{nint}\left(n \sum_{i=1}^{j} p_j \right) - \mathrm{nint}\left(n \sum_{i=1}^{j-1} p_j \right)
\end{equation}
where $\mathrm{nint} : \mathbb R \to \mathbb Z$ is the function which rounds each real number to the nearest integer (with half integers always rounded up).
Using this approximation, we have $\abs{np_j-np'_j} \leq 1$ because each rounding operation in \cref{eq:nint_approx} introduces an error of at most $1/2$ and thus \cref{eq:tv_approx} holds with $C = 1$.
In any case, when \cref{eq:tv_approx} holds for some constant $C\geq 0$, \cref{eq:pinskers} reveals $\kl{p'}{p} \leq \frac{4C}{p_{\min}n^2}$ which yields a useful, albeit crude, lower bound for the multinomial distribution:
\begin{equation}
    \binom{n}{np'_1, \ldots, np'_k} p_1^{np'_1} \cdots p_k^{np'_k} \geq \exp\left(-\frac{4C^2}{p_{\min}n} - k \log (n+1) \right).
\end{equation}

\begin{lem}
    \label{lem:multinomial_dist_lower_bound}
    Let $k\geq 2$ and let $(w_1, \ldots, w_k) \in \mathbb R_{> 0}^{k}$ have minimum $w_{\min} \coloneqq \min\{ w_1, \ldots, w_k\} > 0$ and let $n$ be a positive integer satisfying $n \geq w_{\min}^{-1}$ and define positive integers $(z_1, \ldots, z_k) \in \mathbb N_{>0}^{k}$ by the equation
    \begin{equation}
        z_{i} \coloneqq \ceil{nw_{i}}.
    \end{equation}
    Let $Z = z_1+\cdots+z_k$ and $W = w_1 + \cdots + w_k$ and defined probability distributions $\bar z$ and $\bar w$ by
    \begin{align}
        \bar z_i = \frac{z_i}{Z}, 
        \quad \text{and} \quad 
        \bar w_i = \frac{w_i}{W},
    \end{align}
    for all $i \in \{1,\ldots, k\}$.
    Then
    \begin{equation}
        \label{eq:multinomial_dist_lower_bound}
        \frac{Z!}{z_1! \cdots z_k!}\frac{w_1^{z_{1}}\cdots w_{k}^{z_{k}}}{W^{Z}} \geq e^{-4(k-1)^2}(nW+k+1)^{-k}
    \end{equation}
\end{lem}
\begin{proof}
    Let $\delta_{j} \coloneqq z_j - n w_j$ denote the rounding surplus such that $0\leq \delta_j < 1$.
    Note that for $n \gg k$, we have $Z \approx nW$ because
    \begin{equation}
        nW \leq Z < nW + k.
    \end{equation}
    In this regime, we wish to show that $\bar z_i \approx \bar w_i$.
    \begin{align}
        \abs{\bar z_i - \bar w_i}
        =\abs{\frac{z_i}{Z} - \frac{w_i}{W}} 
        = \frac{\abs{Wz_i - Zw_i}}{ZW}.
    \end{align}
    The numerator above can be bounded independently of $n$ by,
    \begin{align}
        &\abs{Wz_i - Zw_i}\\
        &\qquad= \abs{(nw_i + \delta_i)(w_1+\cdots+w_k) - w_i(\delta_1 + n w_1+ \cdots + \delta_k + n w_k)}, \\
        &\qquad= \abs{\delta_i(w_1+\cdots+\hat w_i+\cdots+w_k) - w_i(\delta_1 + \cdots+\hat \delta_i+\cdots + \delta_k)}, \\
        &\qquad\leq \abs{\delta_i(w_1+\cdots+\hat w_i+\cdots+w_k)} + \abs{w_i(\delta_1 + \cdots+\hat \delta_i+\cdots + \delta_k)}, \\
        &\qquad\leq (w_1+\cdots+\hat w_i+\cdots+w_k) + (k-1)(w_i), \\
        &\qquad\leq (k-1)(w_1+\cdots+w_k), \\
        &\qquad= (k-1)W.
    \end{align}
    Consequently,
    \begin{equation}
        d_{\mathrm{TV}}(\bar z, \bar w) = \max_{1 \leq i \leq k} \abs{\bar z_i - \bar w_i} \leq \frac{k-1}{Z}
    \end{equation}
    and therefore
    \begin{align}
        \frac{Z!}{z_1! \cdots z_k!}\frac{w_1^{z_{1}}\cdots w_{k}^{z_{k}}}{W^{Z}} 
        &\geq (Z+1)^{-k} \exp(-Z\kl{\bar z}{\bar w}), \\
        &\geq (Z+1)^{-k} \exp\left[-\frac{4ZW}{w_{\min}} \left(\frac{k-1}{Z}\right)^2\right], \\
        &\geq (nW+k+1)^{-k} \exp\left[-\frac{4(k-1)^2}{n w_{\min}}\right], \\
        &\geq (nW+k+1)^{-k} \exp(-4(k-1)^2).
    \end{align}
\end{proof}

\begin{lem}
    \label{lem:multinomial_dist_lower_bound_alt}
    Let everything be as in \cref{lem:multinomial_dist_lower_bound}.
    Then
    \begin{equation}
        \label{eq:multinomial_dist_lower_bound_alt}
        \frac{Z!}{z_1! \cdots z_k!}\frac{w_1^{z_{1}}\cdots w_{k}^{z_{k}}}{W^{Z}} \geq \left(\frac{1}{20}\right)^{k}\left(\frac{W}{w_1\cdots w_k}\right)^{\frac{1}{2}}\frac{1}{n^{\frac{3k-1}{2}}}
    \end{equation}
\end{lem}
\begin{proof}
    The proof follows from a version of Stirling's approximation due to \citeauthor{robbins1955remark} \cite{robbins1955remark} which states
    \begin{equation}
        n! = \sqrt{2\pi} n^{n+\frac{1}{2}} e^{-n} e^{r_n}, 
        \qquad
        \text{where}
        \qquad
        \frac{1}{12n+1}\leq r_n \leq \frac{1}{12n}.
    \end{equation}
    Therefore,
    \begin{align}
        \frac{Z!}{z_1! \cdots z_k!}\frac{w_1^{z_{1}}\cdots w_{k}^{z_{k}}}{W^{Z}}
        &=\frac{Z!}{W^{Z}}\frac{w_1^{z_{1}}}{z_1!}\cdots \frac{w_k^{z_{k}}}{z_k!} \\
        &=\left(\frac{1}{2\pi}\right)^{\frac{k-1}{2}}\left(\frac{W}{w_1\cdots w_k}\right)^{\frac{1}{2}}  \left(\frac{Z}{W}\right)^{Z+\frac{1}{2}} \left(\frac{w_1}{z_1}\right)^{z_1+\frac{1}{2}} \cdots \left(\frac{w_k}{z_k}\right)^{z_k+\frac{1}{2}} \exp\left(r_{Z} - \sum_{i} r_{z_i}\right).
    \end{align}
    In this setting, we have assumed $n w_i \geq 1$ which implies
    \begin{equation}
        nw_i \leq z_i < nw_i + 1 \leq 2nw_i,
        \qquad \text{and} \qquad
        nW \leq Z < nW+k < 2nW.
    \end{equation}
    Consequently\footnote{Here we use the identity that $x \geq 1$ implies $\left(\frac{x}{x+1}\right)^{x+\frac{3}{2}} \geq 2^{-5/2}$},
    \begin{equation}
        \left(\frac{Z}{W}\right)^{Z+\frac{1}{2}} \geq n^{nW + \frac{1}{2}}, \qquad
        \left(\frac{w_i}{z_i}\right)^{z_i +\frac{1}{2}} \geq \left(\frac{w_i}{nw_i + 1}\right)^{nw_i + \frac{3}{2}} \geq \frac{2^{-\frac{5}{2}}}{n^{nw_i+\frac{3}{2}}}.
    \end{equation}
    and the remainder term satisfies
    \begin{equation}
        r_{Z} - \sum_{i} r_{z_i} \geq \frac{1}{12Z + 1} - \sum_{i}\frac{1}{12z_i} \geq - \frac{k}{12}.
    \end{equation}
    Altogether, we have
    \begin{align}
        \frac{Z!}{z_1! \cdots z_k!}\frac{w_1^{z_{1}}\cdots w_{k}^{z_{k}}}{W^{Z}}
        &=\left(\frac{1}{2\pi}\right)^{\frac{k-1}{2}}\left(\frac{W}{w_1\cdots w_k}\right)^{\frac{1}{2}} \frac{2^{-\frac{5k}{2}}}{n^{\frac{3k-1}{2}}} \exp\left(-\frac{k}{12}\right).
    \end{align}
    Moreover,
    \begin{equation}
        \left(\frac{1}{2\pi}\right)^{\frac{k-1}{2}}2^{-\frac{5k}{2}} \exp\left(-\frac{k}{12}\right) \geq \left(\frac{1}{20}\right)^{k}
    \end{equation}
\end{proof}

\begin{itemize}
    \item The general subject of multivariate trace inequalities involving exponentials of sums of Hermitian and products of their exponentials is, of course, intimately related. See for instance the paper by \citeauthor{sutter2017multivariate} \cite{sutter2017multivariate}.
    \item Also the article \cite{forrester2014golden} by \citeauthor{forrester2014golden} which describes the connection between Thompson's inequality and hyperbolic geometry quite explicitly.
     \item \citeauthor{friedland2013submodular} considers submodular functions associated to operator monotones evaluated on different principal submatrices of a single Hermitian matrix \cite{friedland2013submodular}. While it is not clear to be yet whether or not one can relate partial sums of matrices to principal submatrices, \cite{friedland2013submodular} references a paper by \citeauthor{bouhtou2008optimization} \cite{bouhtou2008optimization} which proves the following family of inequalities for all positive semidefinite $A,B,C \geq 0$ (and $p \in [0,1]$) \cite[Lemma 5.3]{bouhtou2008optimization}:
        \begin{equation}
            \sum_{j}\lambda_j(A+B)^p + \sum_{j}\lambda_j(B+C)^p \geq \sum_{j}\lambda_j(B)^p + \sum_{j}\lambda_j(A+B+C)^p.
        \end{equation}
    \item \citeauthor{audenaert2010strongly} consider a similar form of ``strongly subadditive'' functions \cite{audenaert2010strongly}
    \item For multivariate inequalities based on idempotents of the symmetric group algebra see \citeauthor{huber2020positive}'s paper \cite{huber2020positive}.
    \item For spectral inequalities derived using semidefinite programming, see \citeauthor{huber2022refuting}'s paper \cite{huber2022refuting}.
    \item \citeauthor{paksoy2014inequalities} produce a large family of polynomial inequalities for positive semidefinite $A,B,C \geq 0$ using the notion of a ``generalized matrix function'' \cite{paksoy2014inequalities} (see also \cite{huber2021matrix,marcus1965generalized})
\end{itemize}

\subsection{Uncategorized}
\begin{itemize}
    \item Elementary introduction to toy models of three-dimensional quantum space \cite{barrett2003geometrical}
    \item \citeauthor{regge2000discrete} \cite{regge2000discrete} a relatively recent review of discrete approaches to gravity. 
        The essential idea, in constrast to the differential geometry of general relativity, is to model curvature discretely instead of continuously. 
        In particular, one can think of a triangulation of the surface of a sphere; the curvature of the sphere is encoded in the triangulation when a configuration of three (or more) triangles meet at a common vertex with sum of interior angles deviating from $\pi$.
        It is in this sense that \citeauthor{regge2000discrete} refers to these types of models as ``piecewise-flat''.
        Contained is an overview of various attempts to generalize discrete theories of three-dimensional gravity to four dimensions \cite[Section III]{regge2000discrete}.
\end{itemize}
